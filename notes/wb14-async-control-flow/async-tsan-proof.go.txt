import (
 "path/filepath"
 "testing"
 "github.com/system-inc/adamic/internal/native"
)

func wb14AsyncTSan(t *testing.T, fixture string, worker bool) {
 t.Helper()
 path, err := filepath.Abs(filepath.Join(repository, "internal/oracle/testdata", fixture))
 if err != nil { t.Fatal(err) }
 truth := onNode(t, path)
 program, err := lowered(t, path)
 if err != nil { t.Fatal(err) }
 code := native.C(program)
 if worker && !emitsParallelMap(code) { t.Fatal("fixture does not call the worker pool") }
 checkParallelCodeUnit(t, code, truth, "tsan", "all", -1)
}
