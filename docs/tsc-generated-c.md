## How to rerun

```sh
S=/tmp/tsc-c   # any new scratch directory
git clone https://github.com/microsoft/TypeScript.git "$S/ts" && git -C "$S/ts" checkout 050880ce59e30b356b686bd3144efe24f875ebc8
(cd stage3/api && npm ci --ignore-scripts)
ADAMIC_TYPESCRIPT_SOURCE="$S/ts" STAGE3_CACHE="$S/cache" bash stage3/apply.sh "$S/adapted" > "$S/apply.log" 2>&1
go build -o "$S/adamic" ./cmd/adamic && go build -o "$S/adamic-meter" ./cmd/adamic-meter
"$S/adamic-meter" --adapt --json "$S/adapted/src/compiler" > "$S/meter.json"
python3 -I review/compiler/tsc-generated-c/measure.py --adamic "$S/adamic" --clang /opt/adamic-tools/llvm/bin/clang \
    --adapted "$S/adapted" --meter "$S/meter.json" --out "$S/out" > "$S/measure.log" 2>&1
cp review/compiler/tsc-generated-c/pristine.txt "$S/out/" && python3 -I review/compiler/tsc-generated-c/report.py "$S/out/rows.json"
```
