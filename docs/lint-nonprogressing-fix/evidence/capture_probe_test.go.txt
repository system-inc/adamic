 target:=filepath.Join(directory,"rules","prefer-promise-reject-errors")
 err:=filepath.WalkDir(source,func(path string,entry os.DirEntry,err error)error{

  if entry.IsDir(){return os.MkdirAll(dest,0755)};data,err:=os.ReadFile(path);if err!=nil{return err};return os.WriteFile(dest,data,0644)
 });if err!=nil{t.Fatal(err)}

  fields:=strings.Split(row,"\t");if fields[1]!="prefer-promise-reject-errors"{continue}
  data,err:=os.ReadFile(fields[0]);if err!=nil{t.Fatal(err)}

 };if !found{t.Fatal("exact Promise recovery row missing")}
}
