package lint

import (
	"os"
	"path/filepath"
	"strings"
	"testing"
)

// Enable this owned artifact as wave12_outside_witness_test.go in a validation checkout.
func TestWave12NestedOutsideWitness(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	oracle := goOracle(t)
	for _, descriptor := range prepareRegistry(t, ".") {
		if descriptor.Name != "nexus/boundary-no-nexus-outside-import" {
			continue
		}
		rows := ownedWitnessRows(t, directory, descriptor)
		if len(rows) != 1 {
			t.Fatalf("expected one nested witness, got %d", len(rows))
		}
		source := strings.SplitN(rows[0], "\t", 2)[0]
		if !strings.Contains(filepath.ToSlash(source), "/libraries/nexus/") {
			t.Fatalf("nested witness path lost: %s", source)
		}
		count := execute(t, "", oracle, "--manifest", manifest(t, rows), "--count")
		if string(count.output) != "1\n" {
			t.Fatalf("upstream nested witness count: %s", count.output)
		}
		negative := filepath.Join(t.TempDir(), "app", "Thing.ts")
		if err := os.MkdirAll(filepath.Dir(negative), 0755); err != nil {
			t.Fatal(err)
		}
		text, err := os.ReadFile(source)
		if err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(negative, text, 0644); err != nil {
			t.Fatal(err)
		}
		off := []string{negative + "\t" + descriptor.Name}
		count = execute(t, "", oracle, "--manifest", manifest(t, off), "--count")
		if string(count.output) != "0\n" {
			t.Fatalf("upstream outside-directory count: %s", count.output)
		}
		rows = append(rows, off...)
		rows = append(rows, source+"\tall")
		compare(t, oracle, buildPort(t, directory, true), directory, manifest(t, rows))
		t.Log("nested outside-import witness: one Go finding inside nexus, none outside; selected and all-rule findings match on source Node, emitted JavaScript and sanitized native")
		return
	}
	t.Fatal("outside-import descriptor missing")
}
