//go:build lintoracle

package main

import (
	"github.com/system-inc/cohere/internal/lint/rule"
	rules "github.com/system-inc/cohere/internal/lint/rules/core"
)

func oracleWave06PreferArrowCallback() rule.Rule { return rules.PreferArrowCallback }
func oracleWave06PreferArrowCallbackOptions(fields []string) any {
	var raw []byte
	if len(fields) > 5 {
		raw = []byte(fields[5])
	}
	options, err := rules.DecodePreferArrowCallbackOptions(raw)
	if err != nil {
		panic(err)
	}
	return options
}
