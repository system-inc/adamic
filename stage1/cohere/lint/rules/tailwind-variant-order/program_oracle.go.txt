	collapse "github.com/system-inc/cohere/internal/lint/rules/tailwind/collapse"
	"github.com/system-inc/cohere/internal/types/program"
	"os"
	"path/filepath"
)

var variantGraph *program.Graph
var variantRoot string

func cleanupVariantProgram() {
	if variantRoot != "" {
		if err := os.RemoveAll(variantRoot); err != nil {
			panic(err)
		}
	}
}
func variantProgram(subject rule.Rule, file *ast.SourceFile) rule.Program {
	if variantGraph == nil {
		root, err := os.MkdirTemp("", "wave104-variant-program-")
		if err != nil {
			panic(err)
		}
		variantRoot = root
		if err = os.MkdirAll(filepath.Join(root, "app/_theme/styles"), 0755); err != nil {
			panic(err)
		}
		for path, text := range map[string]string{"dummy.ts": "export {};\n", "tsconfig.json": `{"compilerOptions":{"strict":true,"target":"ES2022","lib":["ES2022"],"types":[]},"include":["dummy.ts"]}`, "app/_theme/styles/theme.css": `@import "tailwindcss";`} {
			if err = os.WriteFile(filepath.Join(root, path), []byte(text), 0644); err != nil {
				panic(err)
			}
		}
		if err = os.MkdirAll(filepath.Join(root, "node_modules"), 0755); err != nil {
			panic(err)
		}
		packageRoot := os.Getenv("ADAMIC_TAILWIND_PACKAGE")
		if packageRoot == "" {
			panic("ADAMIC_TAILWIND_PACKAGE is required for a non-vacuous variant oracle")
		}
		if err = os.Symlink(packageRoot, filepath.Join(root, "node_modules/tailwindcss")); err != nil {
			panic(err)
		}
		variantGraph, err = program.Build(program.Options{ConfigFileName: filepath.Join(root, "tsconfig.json"), CurrentDirectory: root, SingleThreaded: true})
		if err != nil {
			panic(err)
		}
	}
	return rule.ViewProgram(variantGraph.Program, file, subject)
}
func variantFacts(path, options string) []map[string]any {
	source, err := os.ReadFile(path)
	if err != nil {
		panic(err)
	}
	file := parseFile(path, string(source))
	subject := rules.EnforceConsistentVariantOrder
	view := variantProgram(subject, file)
	result := rules.DesignSystemForProgram(view)
	if result.Err != nil {
		panic(result.Err)
	}
	system := result.System
	settings := rules.DefaultClassLiteralSettings()
	if options != "" && options != "null" {
		var decoded rules.EnforceConsistentVariantOrderOptions
		if err = json.Unmarshal([]byte(options), &decoded); err != nil {
			panic(err)
		}
		if len(decoded.Attributes) > 0 {
			settings.AttributeNames = decoded.Attributes
		}
		if len(decoded.Callees) > 0 {
			settings.CalleeNames = decoded.Callees
		}
		if len(decoded.Variables) > 0 {
			settings.VariablePatterns = decoded.Variables
		}
	}
	reader := rules.NewClassLiteralReader(settings)
	seen := map[string]bool{}
	facts := []map[string]any{}
	var walk func(*ast.Node)
	walk = func(node *ast.Node) {
		for _, literal := range reader.ClassLiteralsIn(node) {
			for _, name := range rules.SplitClasses(literal.Text) {
				if seen[name] {
					continue
				}
				seen[name] = true
				orders := []int{}
				candidates := collapse.ParseCandidate(name, system)
				if len(candidates) > 0 {
					variants := candidates[0].Variants
					rank := system.Variants().BuildVariantOrder(variants)
					for index := len(variants) - 1; index >= 0; index-- {
						v := variants[index]
						order, ok := rank.IndexOf(v)
						if !ok {
							orders = nil
							break
						}
						if globalVariant(v, system) {
							order |= 65536
						}
						orders = append(orders, order)
					}
				}
				if orders == nil {
					orders = []int{}
				}
				facts = append(facts, map[string]any{"class": name, "orders": orders})
			}
		}
		node.ForEachChild(func(child *ast.Node) bool { walk(child); return false })
	}
	walk(file.AsNode())
	return facts
}
func globalVariant(v collapse.ParsedVariant, system *collapse.LoadedDesignSystem) bool {
	if v.Kind != collapse.ParsedVariantKindStatic {
		return false
	}
	switch v.Root {
	case "any-pointer-coarse", "any-pointer-fine", "any-pointer-none", "contrast-less", "contrast-more", "forced-colors", "inverted-colors", "landscape", "motion-reduce", "motion-safe", "noscript", "pointer-coarse", "pointer-fine", "pointer-none", "portrait", "print", "starting":
		return true
	}
	for _, name := range system.Theme().KeysInNamespaces([]string{"--breakpoint"}) {
		if name == v.Root {
			return true
		}
	}
	return false
}
