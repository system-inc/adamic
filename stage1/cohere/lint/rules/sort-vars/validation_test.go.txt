// Not parallel: registry regeneration and captured upstream fixture ledger share scratch paths.
func TestOriginalWitnesses(t *testing.T) {
	var rows []string
	for at, row := range []struct{ source, rule, options string }{
		{"class Foo extends React.Component { UNSAFE_componentWillMount() {} }", "react/no-unsafe", ""},
		{"var b, a;", "sort-vars", ""},
		{"var a, B;", "sort-vars", `{"ignoreCase":true}`},
		{"var c, b = call(), a;", "sort-vars", ""},
		{"const Foo = (React).createReactClass({ ['UNSAFE_componentWillUpdate']: function() {} });", "react/no-unsafe", ""},
	} {
		path := filepath.Join(t.TempDir(), fmt.Sprintf("original-%d.ts", at))
		os.WriteFile(path, []byte(row.source), 0644)
		rows = append(rows, path+"\t"+row.rule+"\t\t\tfalse\t"+row.options)
	}
	compare(t, goOracle(t), buildPort(t, ".", true), ".", manifest(t, rows))
}
func TestOriginalUpstream(t *testing.T) {
	upstream(t)
	data, err := os.ReadFile("/tmp/lint-wave1-10-next/captured.json")
	if err != nil {
		t.Fatal(err)
	}
	var records map[string]struct {
		Rule, File, Source string
		Options            json.RawMessage
	}
	if err = json.Unmarshal(data, &records); err != nil {
		t.Fatal(err)
	}
	var keys []string
	for key := range records {
		keys = append(keys, key)
	}
	sort.Strings(keys)
	var rows []string
	var exclusions []string
	for at, key := range keys {
		row := records[key]
		if row.Rule != "react/no-unsafe" && row.Rule != "sort-vars" {
			continue
		}
		if strings.Contains(row.Source, "<div") || strings.Contains(row.Source, "<Hello") {
			exclusions = append(exclusions, row.Rule+" "+row.Source)
			continue
		}
		ext := filepath.Ext(row.File)
		if ext == "" {
			ext = ".ts"
		}
		path := filepath.Join(t.TempDir(), fmt.Sprintf("original-upstream-%d%s", at, ext))
		os.WriteFile(path, []byte(row.Source), 0644)
		option := string(row.Options)
		if option == "" {
			option = "null"
		}
		rows = append(rows, path+"\t"+row.Rule+"\t\t\tfalse\t"+option)
	}
	os.WriteFile("/tmp/lint-wave1-10-next/original-exclusions.txt", []byte(strings.Join(exclusions, "\n")), 0644)
	t.Logf("original upstream: %d compared, %d JSX parser exclusions", len(rows), len(exclusions))
	compare(t, goOracle(t), buildPort(t, ".", true), ".", manifest(t, rows))
}
func TestOriginalCorpus(t *testing.T) {
	var rows []string
	for _, base := range []string{filepath.Join(os.Getenv("ADAMIC_TYPESCRIPT_SOURCE"), "src/compiler"), filepath.Join(repository, "stage1")} {
		err := filepath.WalkDir(base, func(path string, e fs.DirEntry, err error) error {
			if err != nil {
				return err
			}
			if !e.IsDir() && (strings.HasSuffix(path, ".ts") || strings.HasSuffix(path, ".a")) {
				path, err = filepath.Abs(path)
				if err != nil {
					return err
				}
				for _, rule := range []string{"react/no-unsafe", "react/self-closing-comp", "sort-vars"} {
					rows = append(rows, path+"\t"+rule)
				}
			}
			return nil
		})
		if err != nil {
			t.Fatal(err)
		}
	}
	t.Logf("original corpus: %d files / %d cases", len(rows)/3, len(rows))
	compare(t, goOracle(t), buildPort(t, ".", true), ".", manifest(t, rows))
}

func TestOriginalThroughput(t *testing.T) {
	var paths []string
	filepath.WalkDir(filepath.Join(os.Getenv("ADAMIC_TYPESCRIPT_SOURCE"), "src/compiler"), func(path string, e fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if !e.IsDir() && strings.HasSuffix(path, ".ts") {
			paths = append(paths, path)
		}
		return nil
	})
	oracle := goOracle(t)
	binary := buildPort(t, ".", false)
	for _, rule := range []struct{ name, source string }{
		{"react/no-unsafe", "class Foo extends React.Component { UNSAFE_componentWillMount() {} }\n"},
		{"sort-vars", "var b, a;\n"},
	} {
		fixture := filepath.Join(t.TempDir(), "positive.ts")
		os.WriteFile(fixture, []byte(strings.Repeat(rule.source, 1000)), 0644)
		var rows []string
		for _, path := range append(append([]string{}, paths...), fixture) {
			rows = append(rows, path+"\t"+rule.name)
		}
		manifest := manifest(t, rows)
		var want string
		best := map[string]time.Duration{}
		for round := 0; round < 5; round++ {
			for _, name := range []string{"Go", "Node", "native"} {
				var run execution
				switch name {
				case "Go":
					run = execute(t, "", oracle, "--manifest", manifest, "--count")
				case "Node":
					run = node(t, ".", manifest, true)
				case "native":
					run = execute(t, "", binary, "--manifest", manifest, "--count")
				}
				output := string(run.output)
				if want == "" {
					want = output
				}
				if output != want {
					t.Fatalf("%s count %s != %s", name, output, want)
				}
				if best[name] == 0 || run.duration < best[name] {
					best[name] = run.duration
				}
				t.Logf("%s %s round%d count=%s duration=%s", rule.name, name, round, strings.TrimSpace(output), run.duration)
			}
		}
		var count int
		fmt.Sscan(want, &count)
		for _, name := range []string{"native", "Node", "Go"} {
			t.Logf("%s %s %.2f findings/s (%d, %d files, best of five)", rule.name, name, float64(count)/best[name].Seconds(), count, len(rows))
		}
	}
}

func TestOriginalSelfSnapshots(t *testing.T) {
	source, err := filepath.Abs("rules/react-self-closing-comp/snapshot.a")
	if err != nil {
		t.Fatal(err)
	}
	want, err := os.ReadFile("/tmp/wave10-selfclosing-want.txt")
	if err != nil {
		t.Fatal(err)
	}
	runner, _ := filepath.Abs(filepath.Join(repository, "oracle/node.mjs"))
	for _, variant := range []string{"baseline", "mutant"} {
		path := source
		if variant == "mutant" {
			content, err := os.ReadFile(source)
			if err != nil {
				t.Fatal(err)
			}
			directory := mutant(t, "'fix', ' />'", "'', ' />'", "rules/react-self-closing-comp/rule.a")
			path = filepath.Join(directory, "rules/react-self-closing-comp/snapshot.a")
			text := rewritePortImports(t, "rules/react-self-closing-comp/snapshot.a", string(content))
			if err := os.WriteFile(path, []byte(text), 0644); err != nil {
				t.Fatal(err)
			}
		}
		program, err := load.Load([]string{path})
		if err != nil {
			t.Fatal(err)
		}
		lowered, err := lower.Lower(context.Background(), program)
		if err != nil {
			t.Fatal(err)
		}
		binary := filepath.Join(t.TempDir(), "snapshot")
		if err := native.Build(native.C(lowered), binary, native.Options{Sanitize: true}); err != nil {
			t.Fatal(err)
		}
		js := filepath.Join(t.TempDir(), "snapshot.mjs")
		os.WriteFile(js, []byte(javascript.JavaScript(lowered)), 0644)
		for _, row := range []struct {
			name string
			run  execution
		}{{"Node", execute(t, "", "node", "--disable-warning=ExperimentalWarning", runner, path)}, {"emitted JS", execute(t, "", "node", "--disable-warning=ExperimentalWarning", runner, js)}, {"native", execute(t, "", binary)}} {
			if variant == "baseline" {
				if !bytes.Equal(row.run.output, want) {
					t.Fatalf("%s: %s", row.name, difference(row.run.output, want))
				}
				t.Logf("79 self-closing Go AST snapshots identical on %s: %d bytes", row.name, len(want))
			} else {
				if bytes.Equal(row.run.output, want) {
					t.Fatalf("mutant survived %s", row.name)
				}
				t.Logf("selfclosing_fix_suppressed caught only by output comparison on %s: %s", row.name, difference(row.run.output, want))
			}
		}
	}
}
