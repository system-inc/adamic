// Not parallel: capture and generated registries share process state.
func TestContinuationCorpus(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	rows := upstream(t)
	subjects := []string{"@next/next/no-assign-module-variable", "@typescript-eslint/default-param-last", "structure/tailwind-no-physical-direction"}
	oracle := goOracle(t)
	binary := buildPort(t, directory, true)
	for _, subject := range subjects {
		t.Run(subject, func(t *testing.T) {
			var selected []string
			for _, row := range rows {
				fields := strings.Split(row, "\t")
				if fields[1] != subject {
					continue
				}
				text, err := os.ReadFile(fields[0])
				if err != nil {
					t.Fatal(err)
				}
				if strings.Contains(string(text), "<div className=") {
					t.Logf("EXPLICIT GAP: JSX parser unsupported: %s", row)
					continue
				}
				selected = append(selected, row)
			}
			t.Logf("compared cases: %d", len(selected))
			compare(t, oracle, binary, directory, manifest(t, selected))
		})
	}
}

func TestContinuationThroughput(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	corpus := os.Getenv("ADAMIC_TYPESCRIPT_SOURCE")
	if corpus == "" {
		t.Fatal("missing pinned compiler")
	}
	oracle := goOracle(t)
	binary := buildPort(t, directory, false)
	for _, subject := range []struct{ name, source string }{
		{"@next/next/no-assign-module-variable", "function f%d(){ let module = {}; }\n"},
		{"@typescript-eslint/default-param-last", "function f%d(a = 1, b: number) {}\n"},
		{"structure/tailwind-no-physical-direction", "const c%d = 'ml-4 rtl:mr-4';\n"},
	} {
		var rows []string
		err := filepath.WalkDir(filepath.Join(corpus, "src/compiler"), func(path string, e os.DirEntry, err error) error {
			if err == nil && !e.IsDir() && strings.HasSuffix(path, ".ts") {
				rows = append(rows, path+"\t"+subject.name)
			}
			return err
		})
		if err != nil {
			t.Fatal(err)
		}
		compilerCount := execute(t, "", oracle, "--manifest", manifest(t, rows), "--count")
		t.Logf("%s compiler %d files findings=%s", subject.name, len(rows), strings.TrimSpace(string(compilerCount.output)))
		var source strings.Builder
		for i := 0; i < 1000; i++ {
			fmt.Fprintf(&source, subject.source, i)
		}
		planted := filepath.Join(t.TempDir(), "planted.ts")
		if err := os.WriteFile(planted, []byte(source.String()), 0644); err != nil {
			t.Fatal(err)
		}
		rows = append(rows, planted+"\t"+subject.name)
		path := manifest(t, rows)
		var want []byte
		best := map[string]time.Duration{}
		for round := 0; round < 3; round++ {
			for _, side := range []string{"Go", "native", "Node"} {
				var result execution
				switch side {
				case "Go":
					result = execute(t, "", oracle, "--manifest", path, "--count")
				case "native":
					result = execute(t, "", binary, "--manifest", path, "--count")
				case "Node":
					result = node(t, directory, path, true)
				}
				if want == nil {
					want = result.output
				}
				if !bytes.Equal(want, result.output) {
					t.Fatalf("%s differs on %s", subject.name, side)
				}
				if best[side] == 0 || result.duration < best[side] {
					best[side] = result.duration
				}
				t.Logf("%s round %d %s %.6fs findings=%s", subject.name, round+1, side, result.duration.Seconds(), strings.TrimSpace(string(result.output)))
			}
		}
		var count int
		if _, err := fmt.Sscan(string(want), &count); err != nil || count == 0 {
			t.Fatal("benchmark has no findings")
		}
		for _, side := range []string{"Go", "native", "Node"} {
			t.Logf("%s best of 3 %s %.6fs %.2f findings/s (%d files, %d findings)", subject.name, side, best[side].Seconds(), float64(count)/best[side].Seconds(), len(rows), count)
		}
	}
}

func TestContinuationJSXRefusal(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	// The unchanged Go test asserts one finding for this exact JSX attribute.
	execute(t, filepath.Join(directory, "../../..", "cohere"), "go", "test", "./internal/lint/rules/tailwind", "-run", "^TestNoPhysicalDirectionFires$/a_jsx_attribute$", "-count=1", "-v")
	fixture := filepath.Join(t.TempDir(), "gap.tsx")
	if err := os.WriteFile(fixture, []byte("export const c = <div className=\"flex ml-4\" />;\n"), 0644); err != nil {
		t.Fatal(err)
	}
	path := manifest(t, []string{fixture + "\tstructure/tailwind-no-physical-direction"})
	binary := buildPort(t, directory, true)
	runner, err := filepath.Abs(filepath.Join(repository, "oracle/node.mjs"))
	if err != nil {
		t.Fatal(err)
	}
	for _, side := range []struct {
		name, command string
		args          []string
	}{
		{"Node", "node", []string{"--disable-warning=ExperimentalWarning", runner, filepath.Join(directory, "main.ts"), "--manifest", path}},
		{"native", binary, []string{"--manifest", path}},
	} {
		cmd := exec.Command(side.command, side.args...)
		out, err := os.CreateTemp(t.TempDir(), "stdout-")
		if err != nil {
			t.Fatal(err)
		}
		defer out.Close()
		cmd.Stdout = out
		var stderr bytes.Buffer
		cmd.Stderr = &stderr
		err = cmd.Run()
		status, ok := err.(*exec.ExitError)
		if !ok || status.ExitCode() != 70 || !strings.Contains(stderr.String(), "parser slice") {
			t.Fatalf("%s did not explicitly refuse JSX: %v %s", side.name, err, &stderr)
		}
		t.Logf("%s JSX refusal exit=70: %s", side.name, strings.Split(stderr.String(), "\n")[0])
	}
}
