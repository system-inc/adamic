// Run through a Go overlay into the lint test package; no shared repository file changes.
func TestReactNoTyposPort(t *testing.T) {
	t.Parallel()
	directory := mutant(t, "", "")
	oracle := goOracleFrom(t, directory)
	var rows []string
	count := 0
	for _, row := range upstreamFrom(t, directory) {
		if strings.Split(row, "\t")[1] == "react/no-typos" {
			rows = append(rows, row)
			count++
		}
	}
	t.Logf("react/no-typos captured upstream cases: %d", count)
	for _, row := range generated(t) {
		fields := strings.Split(row, "\t")
		if len(fields) == 1 {
			rows = append(rows, row+"\treact/no-typos")
		} else if fields[1] == "all" {
			fields[1] = "react/no-typos"
			if len(fields) > 5 {
				fields[5] = ""
			} // This rule has no options; the inherited all-rule bag belongs to other rules.
			rows = append(rows, strings.Join(fields, "\t"))
		}
	}
	for _, descriptor := range prepareRegistry(t, directory) {
		if descriptor.Name != "react/no-typos" {
			continue
		}
		witnesses := ownedWitnessRows(t, directory, descriptor)
		for _, row := range witnesses {
			answer := execute(t, "", oracle, "--manifest", manifest(t, []string{row}), "--count")
			if string(answer.output) == "0\n" {
				t.Fatalf("witness reports nothing: %s", row)
			}
		}
		rows = append(rows, witnesses...)
	}
	path := manifest(t, recoveryRows(t, oracle, rows))
	want := execute(t, "", oracle, "--manifest", path).output
	cleanNode := node(t, directory, path, false)
	if diff := difference(cleanNode.output, want); diff != "" {
		t.Fatalf("source Node parity: %s", diff)
	}
	t.Logf("source Node matched %d upstream cases, inherited corpus, and 3 witnesses: %d bytes", count, len(want))
	var change struct{ Name, From, To string }
	data, err := os.ReadFile("rules/react-no-typos/mutant.json")
	if err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(data, &change); err != nil {
		t.Fatal(err)
	}
	module := filepath.Join(directory, "rules/react-no-typos/rule.a")
	data, err = os.ReadFile(module)
	if err != nil {
		t.Fatal(err)
	}
	if strings.Count(string(data), change.From) != 1 {
		t.Fatal("mutation anchor must occur once")
	}
	if err := os.WriteFile(module, []byte(strings.Replace(string(data), change.From, change.To, 1)), 0644); err != nil {
		t.Fatal(err)
	}
	for _, side := range []struct {
		name   string
		result execution
	}{
		{"Node", node(t, directory, path, false)},
		{"emitted JavaScript", emittedNode(t, directory, path, false)},
		{"sanitized native", execute(t, "", buildMutantPort(t, directory), "--manifest", path)},
	} {
		if bytes.Equal(side.result.output, want) {
			t.Fatalf("mutant survived on %s", side.name)
		}
		t.Logf("%s caught on %s: %s", change.Name, side.name, difference(side.result.output, want))
	}
	if err := os.WriteFile(module, data, 0644); err != nil {
		t.Fatal(err)
	}
	t.Log("attempting emitted JavaScript and sanitized native certification")
	compare(t, oracle, buildPort(t, directory, true), directory, path)
}

// Confirm the unchanged base lowers without this registration.
func TestReactNoTyposBaselineControl(t *testing.T) {
	t.Parallel()
	directory := mutant(t, "", "")
	if err := os.RemoveAll(filepath.Join(directory, "rules/react-no-typos")); err != nil {
		t.Fatal(err)
	}
	prepareRegistry(t, directory)
	program, err := load.Load([]string{filepath.Join(directory, "main.ts")})
	if err != nil {
		t.Fatal(err)
	}
	program.EnableTSGo()
	_, err = lower.Lower(context.Background(), program)
	if err != nil {
		t.Fatal(err)
	}
	t.Log("baseline without react/no-typos lowers successfully")
}
