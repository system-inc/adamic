package main

import (
    "context"
    "fmt"
    "os"
    "path/filepath"
    "github.com/system-inc/adamic/internal/load"
    "github.com/system-inc/adamic/internal/lower"
    "github.com/system-inc/adamic/internal/native"
    "github.com/system-inc/adamic/internal/javascript"
)

func main() {
    program, err := load.Load([]string{os.Args[1]})
    if err != nil { panic(err) }
    lowered, err := lower.Lower(context.Background(), program)
    if err != nil { panic(err) }
    directory := os.Args[2]
    if err := os.MkdirAll(directory, 0755); err != nil { panic(err) }
    if err := os.WriteFile(filepath.Join(directory, "driver.mjs"), []byte(javascript.JavaScript(lowered)), 0644); err != nil { panic(err) }
    if err := native.Build(native.C(lowered), filepath.Join(directory, "sanitized"), native.Options{Sanitize:true}); err != nil { panic(err) }
    if err := native.Build(native.C(lowered), filepath.Join(directory, "release"), native.Options{}); err != nil { panic(err) }
    fmt.Println("built emitted JavaScript, ASan/UBSan native and release native")
}
