func completedUpstream(t *testing.T) []string {
	root, err := filepath.Abs(filepath.Join(repository, "cohere"))
	if err != nil {
		t.Fatal(err)
	}
	harness := filepath.Join(root, "internal/lint/testing/rule_testing.go")
	data, err := os.ReadFile(harness)
	if err != nil {
		t.Fatal(err)
	}
	original := "return Result{Diagnostics: diagnostics, SourceFile: sourceFile, capture: captured}"
	replacement := "result := Result{Diagnostics: diagnostics, SourceFile: sourceFile, capture: captured}\n RecordAssertedCase(t, result)\n return result"
	if strings.Count(string(data), original) != 1 {
		t.Fatal("capture overlay anchor changed")
	}
	directory := t.TempDir()
	side := filepath.Join(directory, "rule_testing.go")
	if err := os.WriteFile(side, []byte(strings.Replace(string(data), original, replacement, 1)), 0644); err != nil {
		t.Fatal(err)
	}
	overlay, _ := json.Marshal(map[string]any{"Replace": map[string]string{harness: side}})
	overlayPath := filepath.Join(directory, "overlay.json")
	if err := os.WriteFile(overlayPath, overlay, 0644); err != nil {
		t.Fatal(err)
	}
	capture := filepath.Join(directory, "capture")
	t.Setenv("COHERE_DOCS_CAPTURE", capture)
	selected := map[string]bool{"@typescript-eslint/no-non-null-asserted-optional-chain": true, "@typescript-eslint/no-unnecessary-type-constraint": true}
 execute(t, root, "go", "test", "-overlay="+overlayPath, "./internal/lint/rules/typescript", "-run", "^(TestNoNonNullAssertedOptionalChain|TestNoUnnecessaryTypeConstraint)", "-count=1", "-timeout=10m")
	files, err := filepath.Glob(filepath.Join(capture, "*.jsonl"))
	if err != nil {
		t.Fatal(err)
	}
	type record struct {
		Rule, File, Source, Outcome, FixedSource string
		Options                            json.RawMessage
	}
	unique := map[string]record{}
	for _, path := range files {
		data, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		for _, line := range bytes.Split(data, []byte("\n")) {
			if len(line) == 0 {
				continue
			}
			var row record
			if err := json.Unmarshal(line, &row); err != nil {
				t.Fatal(err)
			}
			if !selected[row.Rule] {
				continue
			}
			key := fmt.Sprintf("%s\t%s\t%+v\t%s", row.Rule, row.File, row.Options, row.Source)
			unique[key] = row
		}
	}
	var keys []string
	for key := range unique {
		keys = append(keys, key)
	}
	sort.Strings(keys)
	var rows []string
	for i, key := range keys {
		row := unique[key]
		path := filepath.Join(directory,fmt.Sprintf("case-%03d",i),strings.TrimPrefix(row.File,"/"))
 if err:=os.MkdirAll(filepath.Dir(path),0755);err!=nil{t.Fatal(err)}
		if err := os.WriteFile(path, []byte(row.Source), 0644); err != nil {
			t.Fatal(err)
		}
		options := "null"
		if len(row.Options) > 0 {
			options = string(row.Options)
		}
		rows = append(rows, fmt.Sprintf("%s\t%s\t\t\tfalse\t%s", path, row.Rule, options))
	}
	if len(rows) < 30 {
		t.Fatalf("capture unexpectedly small: %d cases", len(rows))
	}
	t.Logf("cohere cases: %d unique source/rule/options combinations", len(rows))
	return rows
}
// Not parallel: capture and generated registries share environment state.
func TestCompletedUpstream(t *testing.T) {
 directory,err:=filepath.Abs(".");if err!=nil {t.Fatal(err)}
 rows:=completedUpstream(t)
 compare(t,goOracle(t),buildPort(t,directory,true),directory,manifest(t,rows))
}
func TestCompletedThroughput(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	corpus := os.Getenv("ADAMIC_TYPESCRIPT_SOURCE")
	if corpus == "" {
		t.Fatal("missing pinned TypeScript source")
	}
	oracle := goOracle(t)
	binary := buildPort(t, directory, false)
	for _, subject := range []struct{ name, source string }{
		{"@typescript-eslint/no-non-null-asserted-optional-chain", "const alias%d = foo?.bar!;\n"},
 {"@typescript-eslint/no-unnecessary-type-constraint", "function alias%d<T extends unknown>() {}\n"},
	} {
		var compilerRows []string
		err := filepath.WalkDir(filepath.Join(corpus, "src/compiler"), func(path string, entry os.DirEntry, err error) error {
			if err == nil && !entry.IsDir() && strings.HasSuffix(path, ".ts") {
				compilerRows = append(compilerRows, path+"\t"+subject.name)
			}
			return err
		})
		if err != nil {
			t.Fatal(err)
		}
		compiler := execute(t, "", oracle, "--manifest", manifest(t, compilerRows), "--count")
		t.Logf("%s: compiler alone %d files findings=%s", subject.name, len(compilerRows), strings.TrimSpace(string(compiler.output)))
		var text strings.Builder
		for i := 0; i < 1000; i++ {
			fmt.Fprintf(&text, subject.source, i)
		}
		stress := filepath.Join(t.TempDir(), "throughput.ts")
		if err := os.WriteFile(stress, []byte(text.String()), 0644); err != nil {
			t.Fatal(err)
		}
		rows := append(compilerRows, stress+"\t"+subject.name)
		path := manifest(t, rows)
		best := map[string]time.Duration{}
		var want []byte
		for round := 0; round < 5; round++ {
			for _, side := range []string{"Go", "native", "Node"} {
				var result execution
				if side == "Go" {
					result = execute(t, "", oracle, "--manifest", path, "--count")
				} else if side == "native" {
					result = execute(t, "", binary, "--manifest", path, "--count")
				} else {
					result = node(t, directory, path, true)
				}
				if want == nil {
					want = result.output
				}
				if !bytes.Equal(want, result.output) {
					t.Fatalf("%s count differs on %s", subject.name, side)
				}
				if best[side] == 0 || result.duration < best[side] {
					best[side] = result.duration
				}
				t.Logf("%s round %d %s %.6fs findings=%s", subject.name, round+1, side, result.duration.Seconds(), strings.TrimSpace(string(result.output)))
			}
		}
		var count int
		if _, err := fmt.Sscan(string(want), &count); err != nil || count == 0 {
			t.Fatalf("invalid benchmark count %q", want)
		}
		for _, side := range []string{"Go", "native", "Node"} {
			t.Logf("%s best of 5 %s %.6fs %.2f findings/s (%d files, %d findings)", subject.name, side, best[side].Seconds(), float64(count)/best[side].Seconds(), len(rows), count)
		}
	}
}

// Not parallel: each mutation builds a copied graph and captured fixtures share an environment.
func TestCompletedMutants(t *testing.T) {
 rows:=completedUpstream(t);oracle:=goOracle(t)
 for _,slug:=range []string{"typescript-no-non-null-asserted-optional-chain","typescript-no-unnecessary-type-constraint"} {
 t.Run(slug,func(t *testing.T){
 var change struct{ Name,File,From,To string };data,err:=os.ReadFile(filepath.Join("rules",slug,"mutant.json"));if err!=nil{t.Fatal(err)};if err:=json.Unmarshal(data,&change);err!=nil{t.Fatal(err)}
 path:=manifest(t,rows);want:=execute(t,"",oracle,"--manifest",path).output
 directory:=mutant(t,change.From,change.To,filepath.Join("rules",slug,change.File));binary:=buildPort(t,directory,true)
 for _,side:=range []struct{name string;output []byte}{
 {"source Node",node(t,directory,path,false).output},
 {"emitted JavaScript",completedEmitted(t,directory,path)},
 {"sanitized native",execute(t,"",binary,"--manifest",path).output},
 }{if bytes.Equal(side.output,want){t.Fatalf("%s survived on %s",change.Name,side.name)};t.Logf("%s compiling, successful-exit mutant caught only by output on %s: %s",change.Name,side.name,difference(side.output,want))}
 })
 }
}

func completedEmitted(t *testing.T,directory,path string) []byte {
 program,err:=load.Load([]string{filepath.Join(directory,"main.ts")});if err!=nil{t.Fatal(err)}
 lowered,err:=lower.Lower(context.Background(),program);if err!=nil{t.Fatal(err)}
 module:=filepath.Join(t.TempDir(),"port.mjs");if err:=os.WriteFile(module,[]byte(javascript.JavaScript(lowered)),0644);err!=nil{t.Fatal(err)}
 runner,err:=filepath.Abs(filepath.Join(repository,"oracle/node.mjs"));if err!=nil{t.Fatal(err)}
 return execute(t,"","node","--disable-warning=ExperimentalWarning",runner,module,"--manifest",path).output
}
