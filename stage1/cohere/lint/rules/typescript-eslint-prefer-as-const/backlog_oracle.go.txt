package main

import (
	"bufio"
	"fmt"
	"github.com/microsoft/TypeScript/tsc/shim/ast"
	"github.com/microsoft/TypeScript/tsc/shim/core"
	"github.com/microsoft/TypeScript/tsc/shim/parser"
	"github.com/microsoft/TypeScript/tsc/shim/tspath"
	"github.com/system-inc/cohere/internal/edit"
	"github.com/system-inc/cohere/internal/lint/rule"
	rules "github.com/system-inc/cohere/internal/lint/rules/typescript"
	"os"
	"sort"
	"strings"
	"unicode/utf16"
)

func written(s string) string {
	var out strings.Builder
	for _, r := range s {
		if r >= 32 && r <= 126 && r != 92 {
			out.WriteRune(r)
		} else if r <= 65535 {
			fmt.Fprintf(&out, `\u%04x`, r)
		} else {
			h, l := utf16.EncodeRune(r)
			fmt.Fprintf(&out, `\u%04x\u%04x`, h, l)
		}
	}
	return out.String()
}
func collect(path, source, name string) []rule.Diagnostic {
	kind := core.ScriptKindTS
	if strings.HasSuffix(path, ".tsx") || strings.HasSuffix(path, ".jsx") {
		kind = core.ScriptKindTSX
	}
	file := parser.ParseSourceFile(ast.SourceFileParseOptions{FileName: path, Path: tspath.Path(path)}, source, kind)
	if len(file.Diagnostics()) != 0 {
		panic(fmt.Sprintf("invalid corpus %s: %v", path, file.Diagnostics()))
	}
	var subject rule.Rule
	switch name {
	case "@typescript-eslint/prefer-as-const":
		subject = rules.PreferAsConst
	case "@typescript-eslint/no-unnecessary-type-constraint":
		subject = rules.NoUnnecessaryTypeConstraint
	case "@typescript-eslint/prefer-enum-initializers":
		subject = rules.PreferEnumInitializers
	case "@typescript-eslint/no-extra-non-null-assertion":
		subject = rules.NoExtraNonNullAssertion
	case "@typescript-eslint/no-confusing-non-null-assertion":
		subject = rules.NoConfusingNonNullAssertion
	case "@typescript-eslint/no-unnecessary-parameter-property-assignment":
		subject = rules.NoUnnecessaryParameterPropertyAssignment
	default:
		panic(name)
	}
	var diagnostics []rule.Diagnostic
	ctx := rule.Context{SourceFile: file, FileCache: rule.NewFileCache(), Report: func(d rule.Diagnostic) { diagnostics = append(diagnostics, d) }}
	listeners := subject.Run(ctx, nil)
	var walk func(*ast.Node)
	walk = func(n *ast.Node) {
		if visit := listeners[n.Kind]; visit != nil {
			visit(n)
		}
		n.ForEachChild(func(child *ast.Node) bool { walk(child); return false })
	}
	walk(file.AsNode())
	sort.SliceStable(diagnostics, func(i, j int) bool { return diagnostics[i].Range.Pos() < diagnostics[j].Range.Pos() })
	return diagnostics
}
func apply(source string, fixes []rule.Fix) string {
	ordered := append([]rule.Fix(nil), fixes...)
	sort.SliceStable(ordered, func(i, j int) bool { return ordered[i].Range.Pos() > ordered[j].Range.Pos() })
	previous := len(source)
	for _, fix := range ordered {
		if fix.Range.End() > previous {
			panic("overlapping oracle edits")
		}
		source = source[:fix.Range.Pos()] + fix.Text + source[fix.Range.End():]
		previous = fix.Range.Pos()
	}
	return source
}
func main() {
	data, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
	countOnly := len(os.Args) > 2 && os.Args[2] == "--count"
	count, index := 0, 0
	for _, row := range strings.Split(string(data), "\n") {
		if row == "" {
			continue
		}
		fields := strings.Split(row, "\t")
		raw, err := os.ReadFile(fields[0])
		if err != nil {
			panic(err)
		}
		source := string(raw)
		path := fields[0]
		if len(fields) > 2 {
			path = fields[2]
		}
		diagnostics := collect(path, source, fields[1])
		count += len(diagnostics)
		if !countOnly {
			fmt.Fprintf(out, "case %d\n", index)
			for _, d := range diagnostics {
				fmt.Fprintf(out, "range %d %d %s\t%s\n", d.Range.Pos(), d.Range.End(), d.Message.Id, written(d.Message.Description))
				for _, fix := range d.Fixes {
					fmt.Fprintf(out, "fix %d %d\t%s\n", fix.Range.Pos(), fix.Range.End(), written(fix.Text))
				}
				for _, s := range d.Suggestions {
					fmt.Fprintf(out, "suggestion %s\t%s\n", s.Message.Id, written(s.Message.Description))
					for _, fix := range s.Fixes {
						fmt.Fprintf(out, "edit %d %d\t%s\n", fix.Range.Pos(), fix.Range.End(), written(fix.Text))
					}
					fmt.Fprintf(out, "suggested\t%s\n", written(apply(source, s.Fixes)))
				}
			}
			fixed, err := edit.FixText(path, source, func(file, text string) ([]edit.Proposal, error) {
				return edit.ProposalsFrom(collect(file, text, fields[1])), nil
			}, 10)
			if err != nil || len(fixed.Rejected) != 0 || !fixed.Converged {
				panic(fmt.Sprintf("fix convergence: %v %+v", err, fixed))
			}
			fmt.Fprintf(out, "fixed\t%s\n", written(fixed.Text))
		}
		index++
	}
	if countOnly {
		fmt.Fprintln(out, count)
	}
}
