func TestFifthUpstream(t *testing.T) {
 selected:=map[string]bool{}
 for _,d:=range fifthDescriptors(t){selected[d.Name]=true}
 var rows []string
 counts:=map[string]int{}
 for _,row:=range upstream(t) {
  rule:=strings.Split(row,"\t")[1]
  if selected[rule]{
   var options map[string]json.RawMessage
   fields:=strings.Split(row,"\t")
   if len(fields)>5 && strings.HasPrefix(fields[5],"{"){if err:=json.Unmarshal([]byte(fields[5]),&options);err!=nil{t.Fatal(err)}}
   blocked:=false
   for key,value:=range options {
    if rule=="dot-notation" && strings.EqualFold(key,"allowPattern") && string(value)!=`""` && string(value)!="null" {blocked=true}
    if rule=="id-length" && strings.EqualFold(key,"exceptionPatterns") && string(value)!="[]" && string(value)!="null" {blocked=true}
   }
   if blocked {data,err:=os.ReadFile(fields[0]);if err!=nil{t.Fatal(err)};t.Logf("explicit runtime regex exclusion: %s options=%s source=%q",rule,fields[5],data);continue}
   rows=append(rows,row+"\tapply");counts[rule]++
  }
 }
 t.Logf("fifth upstream cases: %d, per rule %v",len(rows),counts)
 directory,err:=filepath.Abs(".");if err!=nil{t.Fatal(err)}
 oracle:=goOracle(t)
 classified:=execute(t,"",oracle,"--classify",manifest(t,rows)).output
 bad:=map[int]bool{}
 for _,line:=range strings.Split(strings.TrimSpace(string(classified)),"\n") {
  if line==""{continue};var index int;if _,err:=fmt.Sscan(line,&index);err!=nil{t.Fatal(err)};bad[index]=true
 }
 classifier,err:=filepath.Abs("rules/base-consistency-no-bare-throw/classify.mjs");if err!=nil{t.Fatal(err)}
 classifiedNode:=execute(t,"","node","--disable-warning=ExperimentalWarning",classifier,manifest(t,rows)).output
 var failures []struct {Index int;Error string}
 if err:=json.Unmarshal(classifiedNode,&failures);err!=nil{t.Fatal(err)}
 for _,failure:=range failures {bad[failure.Index]=true;parts:=strings.Split(rows[failure.Index],"\t");source,err:=os.ReadFile(parts[0]);if err!=nil{t.Fatal(err)};t.Logf("explicit shared-parser exclusion: %s %s source=%q",parts[1],failure.Error,source)}
 var accepted []string
 for index,row:=range rows {if bad[index] {parts:=strings.Split(row,"\t");source,err:=os.ReadFile(parts[0]);if err!=nil{t.Fatal(err)};t.Logf("explicit parse/fix-engine exclusion: %s source=%q",parts[1],source)} else {accepted=append(accepted,row)}}
 t.Logf("accepted %d upstream cases, excluded %d Go or Adamic parse refusals",len(accepted),len(bad))
 compare(t,oracle,buildPort(t,directory,true),directory,fifthRows(t,oracle,accepted))
}

func TestFifthRegexRefusal(t *testing.T) {
 prepareRegistry(t,".")
 entry,err:=filepath.Abs("rules/typescript-no-non-null-assertion/driver.a");if err!=nil{t.Fatal(err)}
 program,err:=load.Load([]string{entry});if err!=nil{t.Fatal(err)}
 lowered,err:=lower.Lower(context.Background(),program);if err!=nil{t.Fatal(err)}
 binary:=filepath.Join(t.TempDir(),"regex-refusal")
 if err:=native.Build(native.C(lowered),binary,native.Options{Sanitize:true});err!=nil{t.Fatal(err)}
 js:=filepath.Join(t.TempDir(),"regex-refusal.mjs")
 if err:=os.WriteFile(js,[]byte(javascript.JavaScript(lowered)),0644);err!=nil{t.Fatal(err)}
 runner,err:=filepath.Abs(filepath.Join(repository,"oracle/node.mjs"));if err!=nil{t.Fatal(err)}
 for _,row:=range []struct{slug,options string}{{"dot-notation",`{"allowPattern":"^snake_"}`},{"id-length",`{"exceptionPatterns":["^a"]}`}} {
  path:=manifest(t,[]string{ownedWitnesses(t,".",row.slug)[0]+"\t"+row.slug+"\t\t\tfalse\t"+row.options})
  for _,side:=range []struct{name,command string;args []string}{
   {"Node","node",[]string{"--disable-warning=ExperimentalWarning",runner,entry,"--manifest",path,"--count"}},
   {"emitted JavaScript","node",[]string{"--disable-warning=ExperimentalWarning",runner,js,"--manifest",path,"--count"}},
   {"sanitized native",binary,[]string{"--manifest",path,"--count"}},
  } {
   output,err:=os.CreateTemp(t.TempDir(),"refusal-");if err!=nil{t.Fatal(err)}
   command:=exec.Command(side.command,side.args...);command.Stdout=output;command.Stderr=output
   runError:=command.Run();if err:=output.Close();err!=nil{t.Fatal(err)}
   data,err:=os.ReadFile(output.Name());if err!=nil{t.Fatal(err)}
   exit,ok:=runError.(*exec.ExitError)
   want:="adamic: panic: "+row.slug+": runtime regex option requires shared matcher support\n"
   if !ok||exit.ExitCode()!=70||string(data)!=want{t.Fatalf("%s %s refusal: %v %s",row.slug,side.name,runError,data)}
   t.Logf("%s %s: configured runtime regex refuses explicitly, exit 70",row.slug,side.name)
  }
 }
}
