
func wave02Rows(t *testing.T, oracle string, rows []string) []string {
 path := manifest(t, rows)
 excluded := map[int]bool{}
 diagnostics := execute(t, "", oracle, "--classify", path).output
 for _, line := range strings.Split(strings.TrimSpace(string(diagnostics)), "\n") {
  if line == "" { continue }
  fields := strings.SplitN(line, "\t", 2)
  var index int
  if _, err := fmt.Sscan(fields[0], &index); err != nil { t.Fatal(err) }
  excluded[index] = true
  t.Logf("Go parse exclusion %d %s: %s", index, rows[index], fields[1])
 }
 classifier, err := filepath.Abs("rules/structure-tailwind-no-physical-direction/classify.mjs")
 if err != nil { t.Fatal(err) }
 output := execute(t, "", "node", "--disable-warning=ExperimentalWarning", classifier, path).output
 var failures []struct { Index int; Error string }
 if err := json.Unmarshal(output, &failures); err != nil { t.Fatal(err) }
 for _, failure := range failures {
  excluded[failure.Index] = true
  t.Logf("Adamic parse exclusion %d %s: %s", failure.Index, rows[failure.Index], failure.Error)
 }
 var accepted []string
 for index, row := range rows { if !excluded[index] { accepted = append(accepted, row) } }
 t.Logf("coverage: %d accepted, %d excluded, %d total", len(accepted), len(excluded), len(rows))
 return accepted
}

func TestWave02Throughput(t *testing.T) {
 directory, err := filepath.Abs("."); if err != nil { t.Fatal(err) }
 oracle := goOracle(t)
 binary := buildPort(t,directory,false)
 for _, slug := range []string{"structure-tailwind-no-physical-direction","eslint-comments-require-description"} {
  name := "structure/tailwind-no-physical-direction"
  if slug == "eslint-comments-require-description" { name = "@eslint-community/eslint-comments/require-description" }
  source := ownedWitnesses(t,".",slug)[0]
  var rows []string
  for index:=0; index<1000; index++ { rows=append(rows,source+"\t"+name) }
  path:=manifest(t,rows)
  var want []byte
  for _, side := range []struct {name string; run execution}{
   {"Go",execute(t,"",oracle,"--manifest",path,"--count")},
   {"native",execute(t,"",binary,"--manifest",path,"--count")},
   {"Node",node(t,directory,path,true)},
  } {
   if want == nil { want=side.run.output }
   if !bytes.Equal(want,side.run.output) { t.Fatal("benchmark counts differ") }
   var count int
   if _,err:=fmt.Sscan(string(want),&count);err!=nil {t.Fatal(err)}
   t.Logf("%s %s: %d findings, %s, %.2f findings/s (1000 repeated owned witnesses, one run)",name,side.name,count,side.run.duration,float64(count)/side.run.duration.Seconds())
  }
 }
}
