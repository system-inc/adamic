package main

import (
	"encoding/json"
	"fmt"
	core "github.com/system-inc/cohere/internal/lint/rules/core"
	"os"
	"strings"
	"unicode/utf16"
)

func written(s string) string {
	var b strings.Builder
	for _, r := range s {
		if r >= 32 && r <= 126 && r != '\\' {
			b.WriteRune(r)
		} else if r <= 65535 {
			fmt.Fprintf(&b, `\u%04x`, r)
		} else {
			h, l := utf16.EncodeRune(r)
			fmt.Fprintf(&b, `\u%04x\u%04x`, h, l)
		}
	}
	return b.String()
}
func main() {
	data, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	var cases []string
	if err = json.Unmarshal(data, &cases); err != nil {
		panic(err)
	}
	for _, raw := range cases {
		value, err := core.DecodeRequireDescriptionOptions([]byte(raw))
		if err != nil {
			fmt.Println("invalid")
			continue
		}
		fmt.Println("valid")
		options := value.(core.RequireDescriptionOptions)
		for _, v := range options.Ignore {
			fmt.Println("ignore\t" + written(v))
		}
		for _, v := range options.AdditionalDirectives {
			fmt.Println("additional\t" + written(v))
		}
	}
}
