func TestRestrictedTypesMutant(t *testing.T) {
 source:=ownedWitnesses(t,".","typescript-no-restricted-types")[0]
 path:=manifest(t,[]string{source+"\t@typescript-eslint/no-restricted-types\t\t\tfalse\t"+`{"types":{"Foo":false,"string":true}}`})
 want:=execute(t,"",goOracle(t),"--manifest",path).output
 directory:=mutant(t,"ban === undefined || ban.allowed","ban === undefined","rules/typescript-no-restricted-types/rule.a")
 for _,side:=range []struct{name string;run execution}{
  {"Node",node(t,directory,path,false)}, {"emitted JavaScript",emitted(t,directory,path)},
  {"sanitized native",execute(t,"",buildPort(t,directory,true),"--manifest",path)},
 } {if bytes.Equal(side.run.output,want){t.Fatalf("mutant survived on %s",side.name)};t.Logf("allowed ban mutant caught only by output comparison on %s: %s",side.name,difference(side.run.output,want))}
}
func TestRestrictedTypesCompiler(t *testing.T) {
 source:=os.Getenv("ADAMIC_TYPESCRIPT_SOURCE");if source=="" {t.Fatal("missing compiler checkout")}
 var rows []string
 for _,root:=range []string{filepath.Join(source,"src/compiler"),filepath.Join(repository,"stage1")} {
  err:=filepath.WalkDir(root,func(path string,entry fs.DirEntry,err error)error {
   if err!=nil{return err};if !entry.IsDir()&&(strings.HasSuffix(path,".ts")||strings.HasSuffix(path,".a")) {
    absolute,err:=filepath.Abs(path);if err!=nil{return err}
    rows=append(rows,absolute+"\t@typescript-eslint/no-restricted-types\t\t\tfalse\t"+`{"types":{"unknown":true,"object":true,"Function":true,"{}":true,"[]":true}}`)
   };return nil
  });if err!=nil {t.Fatal(err)}
 }
 t.Logf("restricted types compiler/stage1: %d files",len(rows))
 directory,err:=filepath.Abs(".");if err!=nil {t.Fatal(err)}
 compare(t,goOracle(t),buildPort(t,directory,true),directory,manifest(t,rows))
}
func TestRestrictedTypesThroughput(t *testing.T) {

 path:=manifest(t,rows);directory,err:=filepath.Abs(".");if err!=nil {t.Fatal(err)}
 var want []byte
 for _,side:=range []struct{name string;run execution}{
 {"Go",execute(t,"",goOracle(t),"--manifest",path,"--count")},

}
