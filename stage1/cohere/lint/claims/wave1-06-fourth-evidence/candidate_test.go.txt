func wave06Emitted(t *testing.T, directory, path string, count bool) execution {
	t.Helper()
	prepareRegistry(t, directory)
	program, err := load.Load([]string{filepath.Join(directory, "main.ts")})
	if err != nil {
		t.Fatal(err)
	}
	lowered, err := lower.Lower(context.Background(), program)
	if err != nil {
		t.Fatal(err)
	}
	emitted := filepath.Join(t.TempDir(), "main.mjs")
	if err := os.WriteFile(emitted, []byte(javascript.JavaScript(lowered)), 0600); err != nil {
		t.Fatal(err)
	}
	runner, err := filepath.Abs(filepath.Join(repository, "oracle/node.mjs"))
	if err != nil {
		t.Fatal(err)
	}
	args := []string{"--disable-warning=ExperimentalWarning", runner, emitted, "--manifest", path}
	if count {
		args = append(args, "--count")
	}
	return execute(t, "", "node", args...)
}

// Keep filename-dependent declaration-file behavior when replaying upstream cases.
func wave06Cases(t *testing.T) []string {
	t.Helper()
	upstream(t)
	capture := os.Getenv("COHERE_DOCS_CAPTURE")
	files, err := filepath.Glob(filepath.Join(capture, "*.jsonl"))
	if err != nil {
		t.Fatal(err)
	}
	type record struct{ Rule, File, Source string }
	records := map[string]record{}
	for _, file := range files {
		data, err := os.ReadFile(file)
		if err != nil {
			t.Fatal(err)
		}
		for _, line := range bytes.Split(data, []byte("\n")) {
			if len(line) == 0 {
				continue
			}
			var row record
			if err := json.Unmarshal(line, &row); err != nil {
				t.Fatal(err)
			}
			if row.Rule != "@typescript-eslint/no-unsafe-function-type" && row.Rule != "@typescript-eslint/no-useless-empty-export" {
				continue
			}
			records[row.Rule+"\t"+row.File+"\t"+row.Source] = row
		}
	}
	keys := make([]string, 0, len(records))
	for key := range records {
		keys = append(keys, key)
	}
	sort.Strings(keys)
	root := t.TempDir()
	var rows []string
	for i, key := range keys {
		row := records[key]
		directory := filepath.Join(root, fmt.Sprintf("case-%03d", i))
		if err := os.MkdirAll(directory, 0755); err != nil {
			t.Fatal(err)
		}
		path := filepath.Join(directory, filepath.Base(row.File))
		if err := os.WriteFile(path, []byte(row.Source), 0644); err != nil {
			t.Fatal(err)
		}
		rows = append(rows, path+"\t"+row.Rule)
	}
	if len(rows) < 60 {
		t.Fatalf("unexpectedly small own corpus: %d", len(rows))
	}
	t.Logf("filename-preserving upstream cases: %d", len(rows))
	return rows
}

func TestWave06Candidates(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	rows := wave06Cases(t)
	for _, d := range prepareRegistry(t, ".") {
		if d.Slug != "no-unsafe-function-type" && d.Slug != "no-useless-empty-export" {
			continue
		}
		for _, path := range ownedWitnesses(t, directory, d.Slug) {
			rows = append(rows, path+"\t"+d.Name)
		}
	}
	path := manifest(t, rows)
	oracle := goOracle(t)
	want := compare(t, oracle, buildPort(t, directory, true), directory, path)
	emitted := wave06Emitted(t, directory, path, false)
	if diff := difference(emitted.output, want); diff != "" {
		t.Fatalf("emitted JavaScript: %s", diff)
	}
	t.Logf("all four outputs identical: %d bytes", len(want))
}

func TestWave06Mutants(t *testing.T) {
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	oracle := goOracle(t)
	for _, d := range prepareRegistry(t, ".") {
		if d.Slug != "no-unsafe-function-type" && d.Slug != "no-useless-empty-export" {
			continue
		}
		t.Run(d.Slug, func(t *testing.T) {
			var change struct{ Name, File, From, To string }
			data, err := os.ReadFile(filepath.Join("rules", d.Slug, "mutant.json"))
			if err != nil {
				t.Fatal(err)
			}
			if err := json.Unmarshal(data, &change); err != nil {
				t.Fatal(err)
			}
			var rows []string
			for _, path := range ownedWitnesses(t, directory, d.Slug) {
				rows = append(rows, path+"\t"+d.Name)
			}
			path := manifest(t, rows)
			want := execute(t, "", oracle, "--manifest", path).output
			changed := mutant(t, change.From, change.To, filepath.Join("rules", d.Slug, change.File))
			for _, side := range []struct {
				name string
				run  execution
			}{
				{"Node", node(t, changed, path, false)},
				{"emitted JavaScript", wave06Emitted(t, changed, path, false)},
				{"sanitized native", execute(t, "", buildPort(t, changed, true), "--manifest", path)},
			} {
				if bytes.Equal(side.run.output, want) {
					t.Fatalf("mutant survived on %s", side.name)
				}
				t.Logf("%s caught on %s: %s", change.Name, side.name, difference(side.run.output, want))
			}
		})
	}
}

func TestWave06Compiler(t *testing.T) {
	source := os.Getenv("ADAMIC_TYPESCRIPT_SOURCE")
	if source == "" {
		t.Fatal("missing compiler checkout")
	}
	pin := execute(t, "", "git", "-C", source, "rev-parse", "HEAD")
	if strings.TrimSpace(string(pin.output)) != compilerCommit {
		t.Fatal("wrong compiler pin")
	}
	var rows []string
	for _, root := range []string{filepath.Join(source, "src/compiler"), filepath.Join(repository, "stage1")} {
		err := filepath.WalkDir(root, func(path string, entry fs.DirEntry, err error) error {
			if err != nil {
				return err
			}
			if !entry.IsDir() && (strings.HasSuffix(path, ".ts") || strings.HasSuffix(path, ".a")) {
				path, err = filepath.Abs(path)
				if err != nil {
					return err
				}
				for _, rule := range []string{"@typescript-eslint/no-unsafe-function-type", "@typescript-eslint/no-useless-empty-export"} {
					rows = append(rows, path+"\t"+rule)
				}
			}
			return nil
		})
		if err != nil {
			t.Fatal(err)
		}
	}
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("compiler plus stage1: %d file/rule rows", len(rows))
	path := manifest(t, rows)
	want := compare(t, goOracle(t), buildPort(t, directory, true), directory, path)
	emitted := wave06Emitted(t, directory, path, false)
	if diff := difference(emitted.output, want); diff != "" {
		t.Fatal(diff)
	}
}

func TestWave06Throughput(t *testing.T) {
	// Bounded upstream witness throughput includes startup; it is not compiler-corpus throughput.
	directory, err := filepath.Abs(".")
	if err != nil {
		t.Fatal(err)
	}
	oracle := goOracle(t)
	binary := buildPort(t, directory, false)
	for _, d := range prepareRegistry(t, ".") {
		if d.Slug != "no-unsafe-function-type" && d.Slug != "no-useless-empty-export" {
			continue
		}
		var rows []string
		witnesses := ownedWitnesses(t, directory, d.Slug)
		for i := 0; i < 500; i++ {
			for _, path := range witnesses {
				rows = append(rows, path+"\t"+d.Name)
			}
		}
		path := manifest(t, rows)
		var want []byte
		for _, side := range []struct {
			name string
			run  execution
		}{
			{"Go", execute(t, "", oracle, "--manifest", path, "--count")},
			{"native", execute(t, "", binary, "--manifest", path, "--count")},
			{"Node", node(t, directory, path, true)},
		} {
			if want == nil {
				want = side.run.output
			} else if !bytes.Equal(want, side.run.output) {
				t.Fatal("count mismatch")
			}
			var count int
			if _, err := fmt.Sscan(string(want), &count); err != nil {
				t.Fatal(err)
			}
			t.Logf("%s %s: %d findings / %.6fs = %.2f findings/s, %d repeated witness rows", d.Name, side.name, count, side.run.duration.Seconds(), float64(count)/side.run.duration.Seconds(), len(rows))
		}
	}
}
