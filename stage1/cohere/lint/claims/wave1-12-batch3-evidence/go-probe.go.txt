package main

import (
	"fmt"
	"github.com/microsoft/TypeScript/tsc/shim/ast"
	"github.com/microsoft/TypeScript/tsc/shim/core"
	"github.com/microsoft/TypeScript/tsc/shim/parser"
	"github.com/microsoft/TypeScript/tsc/shim/tspath"
	"github.com/system-inc/cohere/internal/lint/rule"
	rules "github.com/system-inc/cohere/internal/lint/rules/next"
	"os"
)

func main() {
	subjects := []rule.Rule{rules.NoBeforeInteractiveScriptOutsideDocument, rules.NoCssTags}
	for i, path := range os.Args[1:] {
		data, err := os.ReadFile(path)
		if err != nil {
			panic(err)
		}
		file := parser.ParseSourceFile(ast.SourceFileParseOptions{FileName: "/Component.tsx", Path: tspath.Path("/Component.tsx")}, string(data), core.ScriptKindTSX)
		if len(file.Diagnostics()) != 0 {
			panic("invalid witness")
		}
		count := 0
		subject := subjects[i]
		ctx := rule.Context{SourceFile: file, FileCache: rule.NewFileCache(), Report: func(d rule.Diagnostic) {
			count++
			fmt.Printf("%s %s %d %d %s\n", subject.Name, d.Message.Id, d.Range.Pos(), d.Range.End(), d.Message.Description)
		}}
		listeners := subject.Run(ctx, nil)
		var walk func(*ast.Node)
		walk = func(n *ast.Node) {
			if visit := listeners[n.Kind]; visit != nil {
				visit(n)
			}
			n.ForEachChild(func(child *ast.Node) bool { walk(child); return false })
		}
		walk(file.AsNode())
		if count != 1 {
			panic(fmt.Sprintf("want one finding, got %d", count))
		}
		fmt.Printf("findings=1 fixes=0 %s\n", subject.Name)
	}
}
