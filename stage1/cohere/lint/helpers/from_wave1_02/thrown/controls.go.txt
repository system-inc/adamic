package control_flow_graph

import "testing"

func TestWave02ThrownControls(t *testing.T) {
	for _, reachable := range []bool{false, true} {
		for _, thrown := range []bool{false, true} {
			for _, final := range []bool{false, true} {
				for _, populated := range []bool{false, true} {
					target := &Block[int]{index: 7, Reachable: reachable, thrown: thrown, final: final}
					builder := &Builder[int]{cur: &Block[int]{index: 3}, finals: []*Block[int]{{index: 1}, {index: 5}}}
					if populated {
						builder.thrown = []*Block[int]{{index: 2}, {index: 4}}
					}
					builder.markThrown(nil)
					builder.markThrown(target)
					builder.markThrown(target)
					target.Reachable = !reachable
					builder.markThrown(target)
				}
			}
		}
	}
}
