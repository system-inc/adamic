package tailwind
import ("fmt"; "unicode/utf16"; "strings")
func Wave08NewTheme(key string) string {
 left,right:=NewTheme(),NewTheme()
 output:=fmt.Sprintf("%t %d %d %d\n",left.Prefix=="",len(left.values),len(left.keyOrder),left.deadKeys)
 values,order:=left.values,left.keyOrder
 values[key]=themeValue{value:key,options:4};order=append(order,key);left.keyOrder=order;left.Prefix="changed";left.deadKeys=2
 _,found:=left.values[key]
 output+=fmt.Sprintf("%t %t %t %d %d %d\n",found,left.keyOrder[0]==key,right.Prefix=="",len(right.values),len(right.keyOrder),right.deadKeys)
 third:=NewTheme()
 return output+fmt.Sprintf("%t %d %d %d\n",third.Prefix=="",len(third.values),len(third.keyOrder),third.deadKeys)
}
func Wave08ClearNamespace(sample string) string {
 output:=""
 for _,namespace:=range []string{"--color","--font","--text","","--grid-row"}{
 for _,options:=range []ThemeOptions{0,1,2,4,8,16,3,31}{
 theme:=NewTheme();keys:=[]string{"--color-a","--colorful-b","--font-a","--font-weight-bold","--font-weightless","--font-size-sm","--text-a","--text-color-a","--grid-row-a","--grid-row-start-a","--color-"+sample,sample}
 for i,key:=range keys{if _,found:=theme.values[key];!found{theme.keyOrder=append(theme.keyOrder,key)};theme.values[key]=themeValue{value:"value",options:ThemeOptions(i%8)}}
 theme.ClearNamespace(namespace,options)
 present:="";for i,key:=range keys{if i>0{present+=","};_,found:=theme.values[key];present+=fmt.Sprint(found)}
 order:="";for i,key:=range theme.keyOrder{if i>0{order+=","};index:=-1;for j,k:=range keys{if key==k{index=j;break}};order+=fmt.Sprint(index)}
 output+=fmt.Sprintf("%d %d %d %s %s\n",len(theme.values),theme.deadKeys,len(theme.keyOrder),present,order)
 }}
 return output
}

func Wave08Width(sample string) string {
 var output strings.Builder
 for _,kind:=range []ParsedVariantKind{"static","functional","compound","arbitrary"}{for _,valueKind:=range []ParsedValueKind{"named","arbitrary"}{for _,hasValue:=range []bool{false,true}{for _,themePresent:=range []bool{false,true}{for _,name:=range []string{sample,"@"+sample}{
 var theme *Theme
 if themePresent {theme=NewTheme();theme.values["--breakpoint-"+sample]=themeValue{value:"40rem"};theme.values["--breakpoint-"+name]=themeValue{value:"40rem"};theme.values["--container-"+sample]=themeValue{value:"24rem"}}
 variant:=ParsedVariant{Kind:kind,Root:name};if hasValue{variant.Value=&ParsedVariantValue{Kind:valueKind,Value:sample}}
 value,found:=resolveBreakpointWidth(theme,variant);codes:=[]string{};for _,unit:=range utf16.Encode([]rune(value)){codes=append(codes,fmt.Sprint(unit))};fmt.Fprintf(&output,"%t %s\n",found,strings.Join(codes,","))
 }}}}}
 return output.String()
}

func Wave08Compare(sample string) string {
 var output strings.Builder
 for mode:=0;mode<8;mode++{for _,themePresent:=range []bool{false,true}{for _,ascending:=range []bool{false,true}{
 var theme *Theme;if themePresent{theme=NewTheme();theme.values["--breakpoint-"+sample]=themeValue{value:"40rem"};theme.values["--container-"+sample]=themeValue{value:"24rem"}}
 left:=ParsedVariant{Kind:ParsedVariantKindFunctional,Root:"min",Value:&ParsedVariantValue{Kind:ParsedValueKindArbitrary,Value:sample}}
 right:=ParsedVariant{Kind:ParsedVariantKindFunctional,Root:"min",Value:&ParsedVariantValue{Kind:ParsedValueKindArbitrary,Value:sample}}
 if mode==1{left.Value.Value="var(--x)"};if mode==2{right.Value.Value="var(--x)"}
 if mode==3{left.Kind=ParsedVariantKindStatic;left.Root=sample;right.Root="@";right.Value.Kind=ParsedValueKindNamed}
 if mode==4||mode==6{right.Kind=ParsedVariantKindCompound};if mode==5||mode==6{left.Kind=ParsedVariantKindCompound}
 if mode==7{left.Value.Kind=ParsedValueKindNamed;right.Root="@";right.Value.Kind=ParsedValueKindNamed}
 fmt.Fprintln(&output,compareBreakpointVariants(theme,left,right,ascending))
 }}}
 return output.String()
}
