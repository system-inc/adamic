package main

import (
	"bufio"
	"encoding/json"
	"fmt"
	"github.com/microsoft/TypeScript/tsc/shim/ast"
	"github.com/microsoft/TypeScript/tsc/shim/core"
	"github.com/microsoft/TypeScript/tsc/shim/parser"
	"github.com/microsoft/TypeScript/tsc/shim/tspath"
	collapse "github.com/system-inc/cohere/internal/lint/rules/tailwind/collapse"
	"os"
	"path/filepath"
	"strings"
	"unicode/utf16"
)

type Input struct{ Rule, File, Source string }
type Case struct {
	Rule  string `json:",omitempty"`
	File  string `json:",omitempty"`
	Value string
}
type Corpus struct {
	Spaces     []int
	Digits     []int
	SuffixSets [][]string
	Cases      []Case
}

func written(value string) string {
	var b strings.Builder
	for _, r := range value {
		if r >= 32 && r <= 126 && r != '\\' {
			b.WriteRune(r)
		} else if r <= 65535 {
			fmt.Fprintf(&b, `\u%04x`, r)
		} else {
			h, l := utf16.EncodeRune(r)
			fmt.Fprintf(&b, `\u%04x\u%04x`, h, l)
		}
	}
	return b.String()
}
func main() {
	if os.Args[1] == "--generate" || (os.Args[1] == "--generate-scan" || os.Args[1] == "--generate-suffix") {
		data, err := os.ReadFile(os.Args[2])
		if err != nil {
			panic(err)
		}
		var inputs []Input
		if err = json.Unmarshal(data, &inputs); err != nil {
			panic(err)
		}
		corpus := Corpus{Spaces: collapse.Wave104JavaScriptSpaces(), Digits: collapse.Wave104Digits()}
		consumers := map[string]bool{"better-tailwindcss/enforce-consistent-class-order": true, "better-tailwindcss/enforce-shorthand-classes": true, "better-tailwindcss/no-conflicting-classes": true, "better-tailwindcss/no-unknown-classes": true}
		for _, input := range inputs {
			if !consumers[input.Rule] {
				continue
			}
			kind := core.ScriptKindTS
			if strings.HasSuffix(input.File, ".tsx") || strings.HasSuffix(input.File, ".jsx") {
				kind = core.ScriptKindTSX
			}
			name, err := filepath.Abs(input.File)
			if err != nil {
				panic(err)
			}
			file := parser.ParseSourceFile(ast.SourceFileParseOptions{FileName: name, Path: tspath.Path(name)}, input.Source, kind)
			var visit func(*ast.Node)
			visit = func(node *ast.Node) {
				switch node.Kind {
				case ast.KindStringLiteral, ast.KindNoSubstitutionTemplateLiteral, ast.KindTemplateHead, ast.KindTemplateMiddle, ast.KindTemplateTail:
					value := node.Text()
					corpus.Cases = append(corpus.Cases, Case{Rule: input.Rule, File: input.File, Value: value})
					for _, part := range strings.Fields(value) {
						corpus.Cases = append(corpus.Cases, Case{Rule: input.Rule, File: input.File, Value: part})
					}
				}
				node.ForEachChild(func(child *ast.Node) bool { visit(child); return false })
			}
			visit(file.AsNode())
		}
		for point := rune(0); point <= 65535; point++ {
			if point >= 0xd800 && point <= 0xdfff {
				continue
			}
			prefix := " "
			if os.Args[1] == "--generate-scan" || os.Args[1] == "--generate-suffix" {
				prefix = "1"
			}
			corpus.Cases = append(corpus.Cases, Case{Value: prefix + string(point) + "X"})
		}
		for _, value := range []string{"", " ", " \t\n\r\v\f", "X ", "\ufeff\u2000X \t", "\u0085X", "\u180eX", "\u200bX", " \U0001f600 X", " \U0010ffffX", "\u2028\u2029"} {
			corpus.Cases = append(corpus.Cases, Case{Value: value})
		}

		if os.Args[1] == "--generate-scan" || os.Args[1] == "--generate-suffix" {
			var words func(string, int)
			words = func(word string, depth int) {
				corpus.Cases = append(corpus.Cases, Case{Value: word})
				if depth == 5 {
					return
				}
				for _, character := range []byte("+-01.eEX") {
					words(word+string(character), depth+1)
				}
			}
			words("", 0)
			for _, number := range []string{"", ".", "+", "-", "0", "1", "10", ".5", "1.5", "5.", "1e", "1e+", "1e-", "1e+5", "1E-5", "+.5", "-.5", "0001", "1e0002", "2.25E+30", "-0", "NaN", "Infinity"} {
				for _, suffix := range []string{"", "x", "%", "px", "em", "é", "😀", " ", "\n", ".", "e+"} {
					corpus.Cases = append(corpus.Cases, Case{Value: number + suffix})
				}
			}
		}
		if os.Args[1] == "--generate-suffix" {
			corpus.SuffixSets = [][]string{nil, {}, {""}, {"%"}, {"px"}, {"em"}, {"deg", "rad"}, {"", "px", "em", "%"}, {"x", "X"}, {"é", "😀", "\u0000"}, {"PX", "EM"}, {"px", "px"}}
		}
		data, err = json.Marshal(corpus)
		if err != nil {
			panic(err)
		}
		if err = os.WriteFile(os.Args[3], data, 0644); err != nil {
			panic(err)
		}
		return
	}
	input := os.Args[1]
	if input == "--scan" || input == "--suffix" {
		input = os.Args[2]
	}
	data, err := os.ReadFile(input)
	if err != nil {
		panic(err)
	}
	var corpus Corpus
	if err = json.Unmarshal(data, &corpus); err != nil {
		panic(err)
	}
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
	if os.Args[1] == "--suffix" {
		ordinal := 0
		for _, c := range corpus.Cases {
			for _, suffixes := range corpus.SuffixSets {
				result, trace := collapse.Wave104Suffix(c.Value, suffixes)
				argument := ""
				if len(trace) > 0 {
					argument = trace[0]
				}
				fmt.Fprintf(out, "%d\t%t\t%d\t%s\n", ordinal, result, len(trace), written(argument))
				ordinal++
			}
		}
		return
	}
	for i, c := range corpus.Cases {
		if os.Args[1] == "--scan" {
			fmt.Fprintf(out, "%d\t%d\n", i, collapse.Wave104ScanNumber(c.Value))
		} else {
			fmt.Fprintf(out, "%d\t%s\n", i, written(collapse.Wave104TrimLeadingJavaScriptSpace(c.Value)))
		}
	}
}
