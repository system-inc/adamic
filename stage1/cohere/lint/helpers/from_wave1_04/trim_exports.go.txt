package tailwind

func Wave104TrimLeadingJavaScriptSpace(value string) string { return trimLeadingJavaScriptSpace(value) }
func Wave104JavaScriptSpaces() []int {
	var spaces []int
	for point := rune(0); point <= 0x10ffff; point++ {
		if isJavaScriptSpace(point) {
			spaces = append(spaces, int(point))
		}
	}
	return spaces
}

func Wave104ScanNumber(value string) int { return scanNumber(value) }
func Wave104Digits() []int {
	var digits []int
	for point := 0; point < 256; point++ {
		if isDigit(byte(point)) {
			digits = append(digits, point)
		}
	}
	return digits
}
