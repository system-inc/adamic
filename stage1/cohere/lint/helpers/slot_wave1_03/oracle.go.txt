package main

import (
	"encoding/json"
	"fmt"
	tailwind "github.com/system-inc/cohere/internal/lint/rules/tailwind/collapse"
	"os"
)

func main() {
	data, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	var entries []tailwind.AdamicEntry
	if err := json.Unmarshal(data, &entries); err != nil {
		panic(err)
	}
	rows := tailwind.AdamicAddCases(entries, os.Args[2])
	if len(os.Args) > 3 && os.Args[3] == "--cases" {
		data, err := json.Marshal(rows)
		if err != nil {
			panic(err)
		}
		fmt.Println(string(data))
		return
	}
	for _, row := range rows {
		fmt.Println(row.Output)
	}
}
