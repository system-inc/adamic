package main

import (
	"encoding/json"
	"fmt"
	"github.com/microsoft/TypeScript/tsc/shim/ast"
	cfg "github.com/system-inc/cohere/internal/lint/ecmascript/control_flow_graph"
	"os"
)

type Row struct{ Chain []int }

func main() {
	raw, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	var rows []Row
	if err = json.Unmarshal(raw, &rows); err != nil {
		panic(err)
	}
	factory := ast.NewNodeFactory(ast.NodeFactoryHooks{})
	for i, row := range rows {
		var node *ast.Node
		for j := len(row.Chain) - 1; j >= 0; j-- {
			kind := ast.Kind(row.Chain[j])
			if kind == ast.KindParenthesizedExpression {
				node = factory.NewParenthesizedExpression(node)
			} else if kind == ast.KindBinaryExpression {
				node = factory.NewBinaryExpression(nil, &ast.Node{Kind: ast.KindIdentifier}, nil, &ast.Node{Kind: ast.KindEqualsToken}, &ast.Node{Kind: ast.KindIdentifier})
			} else {
				node = &ast.Node{Kind: kind}
			}
		}
		fmt.Printf("case %d\nresult\t%t\n", i, cfg.Wave11DestructuringTarget(node))
	}
}
