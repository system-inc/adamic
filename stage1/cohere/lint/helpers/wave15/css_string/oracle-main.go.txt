package main

import (
	"bufio"
	"encoding/hex"
	"encoding/json"
	"fmt"
	rules "github.com/system-inc/cohere/internal/lint/rules/tailwind/collapse"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
)

func encode(value string) string { return hex.EncodeToString([]byte(value)) }
func decode(value string) string {
	b, e := hex.DecodeString(value)
	if e != nil {
		panic(e)
	}
	return string(b)
}
func main() {
	if os.Args[1] == "generate" {
		output, e := os.Create(os.Args[2])
		if e != nil {
			panic(e)
		}
		defer output.Close()
		coverage := map[string]int{}
		stems := []string{"enforce_canonical_classes", "enforce_consistent_class_order", "enforce_consistent_variant_order", "enforce_shorthand_classes", "no_conflicting_classes", "no_unknown_classes"}
		for _, stem := range stems {
			paths, e := filepath.Glob(filepath.Join(os.Args[3], stem+"*_test.go"))
			if e != nil {
				panic(e)
			}
			set := map[string]bool{}
			for _, path := range paths {
				f, e := parser.ParseFile(token.NewFileSet(), path, nil, 0)
				if e != nil {
					panic(e)
				}
				ast.Inspect(f, func(n ast.Node) bool {
					lit, ok := n.(*ast.BasicLit)
					if ok && lit.Kind == token.STRING {
						value, e := strconv.Unquote(lit.Value)
						if e != nil {
							panic(e)
						}
						set[value] = true
					}
					return true
				})
			}
			keys := []string{}
			for k := range set {
				keys = append(keys, k)
			}
			sort.Strings(keys)
			coverage[stem] = len(keys)
			for _, root := range keys {
				for _, quote := range []byte{34, 39} {
					input := string(quote) + root + string(quote)
					fmt.Fprintf(output, "%s\t0\t%d\n", encode(input), quote)
				}
				for position := 0; position < len(root); position++ {
					if root[position] == 34 || root[position] == 39 {
						fmt.Fprintf(output, "%s\t%d\t%d\n", encode(root), position, root[position])
					}
				}
			}
		}
		for quote := 0; quote < 256; quote++ {
			for value := 0; value < 256; value++ {
				for _, input := range []string{string([]byte{byte(quote), byte(value), byte(quote)}), string([]byte{byte(quote), 92, byte(value), byte(quote)}), string([]byte{0, 255, byte(quote), byte(value), 59, 13, 10})} {
					start := 0
					if input[0] == 0 && len(input) == 7 {
						start = 2
					}
					fmt.Fprintf(output, "%s\t%d\t%d\n", encode(input), start, quote)
				}
			}
		}
		for _, input := range []string{"", "'", "'unterminated", "'x\rtext'", "'x;\n", "'x\r\n", "'x\\\ntext'", "é'🌍;\n"} {
			for start := 0; start <= len(input); start++ {
				fmt.Fprintf(output, "%s\t%d\t39\n", encode(input), start)
			}
		}
		data, e := json.MarshalIndent(coverage, "", "  ")
		if e != nil {
			panic(e)
		}
		if e = os.WriteFile(os.Args[2]+".coverage.json", data, 0644); e != nil {
			panic(e)
		}
		return
	}
	file, e := os.Open(os.Args[1])
	if e != nil {
		panic(e)
	}
	defer file.Close()
	scanner := bufio.NewScanner(file)
	scanner.Buffer(make([]byte, 4096), 16*1024*1024)
	for scanner.Scan() {
		fields := strings.Split(scanner.Text(), "\t")
		input := decode(fields[0])
		start, e := strconv.Atoi(fields[1])
		if e != nil {
			panic(e)
		}
		quote, e := strconv.Atoi(fields[2])
		if e != nil {
			panic(e)
		}
		index, failed, offset, message := rules.Wave15CSSString(input, start, byte(quote))
		fmt.Printf("%d\t%t\t%d\t%s.\n", index, failed, offset, encode(message))
	}
	if e = scanner.Err(); e != nil {
		panic(e)
	}
}
