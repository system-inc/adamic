package tailwind

// Own oracle seam; original Get and VariantKind run unchanged.
func Wave15VariantKind(registrations map[string]string, root string) string {
	registry := NewVariantRegistry()
	for key, kind := range registrations {
		registry.Register(key, ParsedVariantKind(kind))
	}
	system := LoadedDesignSystem{variants: registry}
	return string(system.VariantKind(root))
}
