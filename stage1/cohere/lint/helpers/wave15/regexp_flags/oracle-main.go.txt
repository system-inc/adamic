func encode(s string) string {
 var out strings.Builder
 for _, c := range utf16.Encode([]rune(s)) { fmt.Fprintf(&out, "%04x", c) }
 return out.String()
}
func decode(s string) string {
 units := []uint16{}
 for i:=0; i<len(s); i+=4 { n,e:=strconv.ParseUint(s[i:i+4],16,16); if e!=nil {panic(e)};units=append(units,uint16(n)) }
 return string(utf16.Decode(units))
}
