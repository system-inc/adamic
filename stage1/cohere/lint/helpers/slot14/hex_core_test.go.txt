package core

import (
	rule_testing "github.com/system-inc/cohere/internal/lint/testing"
	"testing"
)

func TestSlot14HexExports(t *testing.T) {
	result := runNoRestrictedExports(t, noRestrictedExportsCase{
		sourceText: "export const foo = 1;",
		options:    NoRestrictedExportsOptions{RestrictedNamedExportsPattern: `^\x66oo$`},
	})
	rule_testing.ExpectFindings(t, result, "restrictedNamed")
}
func TestSlot14HexImports(t *testing.T) {
	result := runNoRestrictedImports(t, noRestrictedImportsCase{
		sourceText: "import value from 'foo';",
		options:    NoRestrictedImportsOptions{Patterns: []NoRestrictedImportsPattern{{Regex: `^\u0066oo$`}}},
	})
	rule_testing.ExpectFindings(t, result, "patterns")
}
