package regexp

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"testing"
)

type slot14OracleUnicodeInput struct {
	Source                  []int
	Index, Size, Digits     int
	Unicode, InClass, Named bool
	Groups                  int
}

func TestSlot14UnicodeEscape(t *testing.T) {
	data, err := os.ReadFile(os.Getenv("SLOT14_UNICODE_INPUTS"))
	if err != nil {
		t.Fatal(err)
	}
	var rows []slot14OracleUnicodeInput
	if err := json.Unmarshal(data, &rows); err != nil {
		t.Fatal(err)
	}
	output, err := os.Create(os.Getenv("SLOT14_UNICODE_OUTPUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer output.Close()
	for _, row := range rows {
		bytes := make([]byte, len(row.Source))
		for at, value := range row.Source {
			if value < 0 || value > 255 {
				t.Fatal("byte outside its domain")
			}
			bytes[at] = byte(value)
		}
		result, err := decodeUnicodeEscape(string(bytes), row.Index, row.Size, escapeContext{unicode: row.Unicode, inClass: row.InClass, groups: row.Groups, named: row.Named})
		text := ""
		if err != nil {
			text = err.Error()
		}
		fmt.Fprintf(output, "%d\t%d\t%d\t%d\t%t\t%t\t%t\t%x\n", result.kind, result.set, result.r, result.width, result.negated, err != nil, errors.Is(err, ErrUnsupportedSyntax), []byte(text))
	}
}
