package regexp

import (
	"encoding/json"
	"os"
	"sync"
)

type slot14UnicodeInput struct {
	Source                  []int
	Index, Size, Digits     int
	Unicode, InClass, Named bool
	Groups                  int
}

var slot14UnicodeMutex sync.Mutex

func decodeUnicodeEscape(source string, index, size int, context escapeContext) (decodedEscape, error) {
	result, err := slot14OriginalUnicodeEscape(source, index, size, context)
	path := os.Getenv("SLOT14_UNICODE_CAPTURE")
	if path == "" {
		return result, err
	}
	bytes := make([]int, len(source))
	for at := 0; at < len(source); at++ {
		bytes[at] = int(source[at])
	}
	row := slot14UnicodeInput{Source: bytes, Index: index, Size: size, Unicode: context.unicode, InClass: context.inClass, Named: context.named, Groups: context.groups}
	data, marshalErr := json.Marshal(row)
	if marshalErr != nil {
		panic(marshalErr)
	}
	slot14UnicodeMutex.Lock()
	defer slot14UnicodeMutex.Unlock()
	file, openErr := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0644)
	if openErr != nil {
		panic(openErr)
	}
	if _, writeErr := file.Write(append(data, '\n')); writeErr != nil {
		panic(writeErr)
	}
	if closeErr := file.Close(); closeErr != nil {
		panic(closeErr)
	}
	return result, err
}
