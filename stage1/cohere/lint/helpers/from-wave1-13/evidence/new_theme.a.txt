export interface ThemeValue {
    value: string;
    options: number;
}
export interface MutableThemeState {
    prefix: string;
    values: Map<string, ThemeValue>;
    keyOrder: string[];
    deadKeys: number;
}
// Keep the same store shape as the independently owned Theme mutation helpers.
export function newTheme(): MutableThemeState {
    return { prefix: '', values: new Map<string, ThemeValue>(), keyOrder: [], deadKeys: 0 };
}
