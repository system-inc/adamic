package checker

import (
	"fmt"
	"strings"

	"github.com/microsoft/TypeScript/tsc/shim/ast"
	"github.com/microsoft/TypeScript/tsc/shim/checker"
)

// This question returns raw symbol facts without rendering a declaration's parent
// name. Computed names are AST nodes, not text-bearing identifier nodes.
// The optional read variant selects the symbol a shorthand or local export reads.
func (p *Program) inspectCallbackSymbolFacts(out *fields, c *checker.Checker, node *ast.Node, question string) (string, error) {
	if strings.Split(question, "\n")[0] != "callback-symbol-facts" {
		return p.inspectTypeAliasInfo(out, c, node, question)
	}
	if false {
		return "", fmt.Errorf("callback-symbol-facts requires an Identifier and optional read variant")
	}
	symbol := c.GetSymbolAtLocation(node)
	if question == "callback-symbol-facts\nread" && node.Parent != nil {
		switch node.Parent.Kind {
		case ast.KindShorthandPropertyAssignment:
			if node.Parent.Name() == node {
				symbol = c.GetShorthandAssignmentValueSymbol(node.Parent)
			}
		case ast.KindExportSpecifier:
			symbol = c.GetExportSpecifierLocalTargetSymbol(node.Parent)
		}
	}
	out.number(p.symbolID(symbol))
	count := 0
	if symbol != nil {
		count = len(symbol.Declarations)
	}
	out.number(uint64(count))
	if symbol != nil {
		for _, declaration := range symbol.Declarations {
			source := ast.GetSourceFileOfNode(declaration)
			if source == nil {
				return "", fmt.Errorf("symbol declaration has no source")
			}
			out.yes(source.IsDeclarationFile)
		}
	}
	return out.String(), nil
}
