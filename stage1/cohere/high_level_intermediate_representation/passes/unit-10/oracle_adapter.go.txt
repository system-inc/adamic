type unit10Dependency struct {
	Root struct {
		Identifier uint32
		Global     bool
		Name       string
	}
	Path []DependencyPathEntry
}

func (d unit10Dependency) value() ManualMemoDependency {
	return ManualMemoDependency{Root: ManualMemoRoot{IsGlobal: d.Root.Global, Name: d.Root.Name,
		Place: Place{Identifier: static_single_assignment.IdentifierId(d.Root.Identifier)}}, Path: d.Path}
}
func TestUnit10DependencyCheckpoint(t *testing.T) {
	t.Parallel()
	data, err := os.ReadFile(os.Getenv("HIR_UNIT10_INPUT"))
	if err != nil {
		t.Fatal(err)
	}
	var cases []struct {
		Name             string
		Inferred, Source unit10Dependency
	}
	if err := json.Unmarshal(data, &cases); err != nil {
		t.Fatal(err)
	}
	// The input is Go's state immediately before CompareManualMemoDependencies.
	if err := os.WriteFile(os.Getenv("HIR_UNIT10_BEFORE"), data, 0600); err != nil {
		t.Fatal(err)
	}
	var after strings.Builder
	for _, c := range cases {
		result := CompareManualMemoDependencies(c.Inferred.value(), c.Source.value())
		fmt.Fprintf(&after, "%s\t%d\t%s\n", c.Name, result, result.String())
	}
	for first := CompareDependencyOk; first <= CompareDependencyRefAccessDifference; first++ {
		for second := CompareDependencyOk; second <= CompareDependencyRefAccessDifference; second++ {
			fmt.Fprintf(&after, "merge\t%d\t%d\t%d\n", first, second, MergeCompareDependencyResults(first, second))
		}
	}
	fmt.Fprintln(&after, CompareDependencyResult(99).String())
	if err := os.WriteFile(os.Getenv("HIR_UNIT10_AFTER"), []byte(after.String()), 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("comparison before/after: %d path cases, 25 merge cases, 1 message case", len(cases))
}

func TestUnit10ScopeProducerCheckpoint(t *testing.T) {
	t.Parallel()
	// This literal array graph needs no checker.
	source := parser.ParseSourceFile(ast.SourceFileParseOptions{FileName: "/test.tsx", PathKey: "/test.tsx"}, "function f(){return[]}", core.ScriptKindTSX)
	var node *ast.Node
	source.AsNode().ForEachChild(func(n *ast.Node) bool {
		if ast.IsFunctionLike(n) {
			node = n
			return true
		}
		return false
	})
	if node == nil {
		t.Fatal("missing function")
	}
	function := Lower(node, nil)
	Construct(function)
	OutlineFunctions(function)
	InferReactive(function, nil)
	DropManualMemoization(function)
	if InlineImmediatelyInvokedFunctionExpressionsIncludingMemoCallbacks(function) > 0 {
		MergeConsecutiveBlocks(function)
	}
	EliminateDeadCode(function)
	ranges := InferMutableRanges(function)
	sets := FindDisjointMutableValuesWithRanges(function, ranges)
	scopes := AssignReactiveScopesWithSets(function, ranges, sets)
	aligned, merged := AlignThenMergeReactiveScopes(function, scopes)
	BuildReactiveScopeTerminals(function, scopes, MergedScopeIdentity{Aligned: aligned, Merged: merged})
	found := false
	for _, block := range function.Blocks {
		if _, ok := block.Terminal.(*Scope); ok {
			found = true
		}
	}
	if !found {
		t.Fatal("reproducer no longer emits Scope")
	}
	if err := os.WriteFile(os.Getenv("HIR_UNIT10_SCOPE"), []byte(oracleDump(function)), 0600); err != nil {
		t.Fatal(err)
	}
	t.Log("actual Go scope producer emits the terminal missing from shared replay")
}
