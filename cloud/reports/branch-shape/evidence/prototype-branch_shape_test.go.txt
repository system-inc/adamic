// The runtime inputs include every value a lossy conversion would confuse with a case,
// and cases with effects must retain their original lazy evaluation order.
func TestNumericSwitchMatchesNode(t *testing.T) {
	t.Parallel()
	const source = `function select(value: number): string {
    switch (value) {
        case 4: return 'four';
        case 0: return 'zero';
        case 1: return 'one';
        case 2: return 'two';
        case 3: return 'three';
        case 2147483647: return 'maximum';
        case 5: return 'five';
        case 0: return 'duplicate';
        default: return 'other';
    }
}
for (const value of [-2, 0, -0, 1, 2, 3, 2147483647, -2147483648, 1.5, -1.5, 2147483648, -2147483649, NaN, Infinity, -Infinity, 1e100]) {
    console.log(select(value));
}
let result = '';
for (let index = 0; index < 5; index += 1) {
    switch (index) {
        case 0: result += 'a'; break;
        case 1: continue;
        case 2: { const label = 'x'.repeat(index); result += label; break; }
        case 3: result += 'b'; break;
        default: result += 'c'; break;
    }
}
console.log(result);
`
	directory := t.TempDir()
	path := filepath.Join(directory, "switch.a")
	if err := os.WriteFile(path, []byte(source), 0o644); err != nil {
		t.Fatal(err)
	}
	loaded, err := load.Load([]string{path})
	if err != nil {
		t.Fatal(err)
	}
	program, err := lower.Lower(context.Background(), loaded)
	if err != nil {
		t.Fatal(err)
	}
	generated := C(program)
	if !strings.Contains(generated, "switch (") {
		t.Fatal("numeric cases still emit comparison chains")
	}
	want := runWithInput(t, source, "node", "--input-type=module-typescript")
	for _, sanitize := range []bool{false, true} {
		binary := filepath.Join(directory, "switch")
		if err := Build(generated, binary, Options{Sanitize: sanitize}); err != nil {
			t.Fatal(err)
		}
		if got := runWithInput(t, "", binary); got != want {
			t.Fatalf("sanitize %v: native %q; Node %q", sanitize, got, want)
		}
	}
}

func TestScalarLogicalMatchesNode(t *testing.T) {
	t.Parallel()
	const source = `function range(value: number): boolean {
    return (value >= 65 && value <= 90) || (value >= 97 && value <= 122) || value === 36 || value === 95;
}
function nested(a: boolean, b: boolean, c: boolean): boolean { return (a && b) || (!a && c); }
for (const value of [0, 36, 64, 65, 90, 91, 95, 97, 122, 123, 65.5, NaN, Infinity, -Infinity]) {
    console.log(range(value).toString());
}
for (const a of [false, true]) {
    for (const b of [false, true]) {
        for (const c of [false, true]) { console.log(nested(a, b, c).toString()); }
    }
}
let calls = 0;
function effect(): boolean { calls += 1; return true; }
function guarded(value: boolean): boolean { return value && effect(); }
function guardedOr(value: boolean): boolean { return value || effect(); }
console.log(guarded(false).toString());
console.log(guardedOr(true).toString());
console.log(calls.toString());
console.log(guarded(true).toString());
console.log(calls.toString());
function missing(): { value: number } | undefined { return undefined; }
const object = missing();
console.log((object !== undefined && object.value > 0).toString());
`
	directory := t.TempDir()
	path := filepath.Join(directory, "logical.a")
	if err := os.WriteFile(path, []byte(source), 0o644); err != nil {
		t.Fatal(err)
	}
	loaded, err := load.Load([]string{path})
	if err != nil {
		t.Fatal(err)
	}
	program, err := lower.Lower(context.Background(), loaded)
	if err != nil {
		t.Fatal(err)
	}
	generated := C(program)
	if !strings.Contains(generated, " & ") || !strings.Contains(generated, " | ") {
		t.Fatal("total scalar comparisons still short-circuit")
	}
	want := runWithInput(t, source, "node", "--input-type=module-typescript")
	for _, sanitize := range []bool{false, true} {
		binary := filepath.Join(directory, "logical")
		if err := Build(generated, binary, Options{Sanitize: sanitize}); err != nil {
			t.Fatal(err)
		}
		if got := runWithInput(t, "", binary); got != want {
			t.Fatalf("sanitize %v: native %q; Node %q", sanitize, got, want)
		}
	}
}
