func TestImmortalStaticDispatch(t *testing.T) {
	t.Parallel()
	for _, value := range []string{"&adamic_string_0", "&adamic_string_184", "&adamic_string_empty", "&adamic_box_true", "&adamic_box_false.heap", "&adamic_typeof_number", "((adamic_heap *)(&adamic_string_4))"} {
		if retained(value) != value {
			t.Fatalf("retained static %s", value)
		}
		e := &emitter{program: &ir.Program{GraphTypes: map[int]bool{1: true}}}
		if got := e.heldReferenceIn("holder", value); got != value {
			t.Fatalf("graph held immortal: %s", got)
		}
		e.dropIn("holder", value)
		if e.out.Len() != 0 {
			t.Fatal("graph dropped immortal")
		}
	}
	for _, value := range []string{"adamic_temporary_1", "&adamic_string_fake", "&adamic_string_1 + 1", "heap"} {
		if retained(value) != "adamic_retain("+value+")" {
			t.Fatalf("unproven retain omitted: %s", value)
		}
		e := &emitter{program: &ir.Program{GraphTypes: map[int]bool{1: true}}}
		if got := e.heldReferenceIn("holder", value); !strings.HasPrefix(got, "adamic_graph_hold(") {
			t.Fatalf("dispatch changed: %s", got)
		}
	}
}

const immortalFieldsSource = `class Label {
 spelling: string = 'first';
 heapLabel: string = 'initial';
 set(): void { this.spelling = 'second'; }
 heap(text: string): void { this.heapLabel = text; }
}
const label = new Label();
label.set();
console.log(label.spelling);
for (let index = 1; index < 6; index += 1) {
 label.heap('runtime'.repeat(index));
}
console.log(label.heapLabel);
label.heap('final');
console.log(label.heapLabel);
`

func immortalFieldsProgram(t *testing.T) *ir.Program {
	t.Helper()
	path := filepath.Join(t.TempDir(), "immortal.a")
	if err := os.WriteFile(path, []byte(immortalFieldsSource), 0644); err != nil {
		t.Fatal(err)
	}
	loaded, err := load.Load([]string{path})
	if err != nil {
		t.Fatal(err)
	}
	program, err := lower.Lower(context.Background(), loaded)
	if err != nil {
		t.Fatal(err)
	}
	return program
}

// Semantic lane has no code-shape assertion: a mutant must reach the sanitizer.
func TestImmortalFieldsMatchesNode(t *testing.T) {
	t.Parallel()
	program := immortalFieldsProgram(t)
	want := runWithInput(t, immortalFieldsSource, "node", "--input-type=module-typescript")
	generated := C(program)
	for _, sanitize := range []bool{false, true} {
		binary := filepath.Join(t.TempDir(), "immortal")
		if err := Build(generated, binary, Options{Sanitize: sanitize}); err != nil {
			t.Fatal(err)
		}
		if got := runWithInput(t, "", binary); got != want {
			t.Fatalf("sanitize %v: %q != Node %q", sanitize, got, want)
		}
	}
}

func TestImmortalFieldProof(t *testing.T) {
	t.Parallel()
	e := &emitter{program: immortalFieldsProgram(t)}
	if !e.immortalField("spelling") {
		t.Fatal("literal-only writes not proven")
	}
	if e.immortalField("heapLabel") {
		t.Fatal("heap write accepted as immortal")
	}
}
