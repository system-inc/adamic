// #0d56xpd question 2: the undefined! placeholder parent taken out by destructuring, as accessKind takes
// it (const { parent } = node), then read with a plain dot.
interface N {
	readonly kind: number;
	readonly parent: N;
}
const root: N = { kind: 1, parent: undefined! };
function parentKind(node: N): string {
	const { parent } = node;
	return `${parent.kind}`;
}
console.log('before');
console.log(parentKind(root));
console.log('after');
