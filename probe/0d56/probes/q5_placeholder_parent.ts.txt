// #0d56xpd question 5: the root's parent left as the undefined! placeholder, as tsc's Node constructor
// leaves a SourceFile's, and read the way accessKind reads it: const { parent } = node; switch (parent?.kind).
interface N {
	readonly kind: number;
	readonly parent: N;
}
const root: N = { kind: 1, parent: undefined! };
const child: N = { kind: 2, parent: root };
function accessKind(node: N): string {
	const { parent } = node;
	switch (parent?.kind) {
		case 1:
			return 'one';
		default:
			return 'other';
	}
}
console.log(accessKind(child));
console.log(accessKind(root));
