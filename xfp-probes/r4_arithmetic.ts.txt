enum E {
	A,
	B,
	C,
}
function describe(value: E): string {
	switch (value) {
		case E.A:
			return "a";
		case E.B:
			return "b";
		case E.C:
			return "c";
		default: {
			const unreachable: never = value;
			return "unreachable";
		}
	}
}
