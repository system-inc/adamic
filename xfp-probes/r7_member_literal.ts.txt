// #xfpsqtn soundness road: a number cast into a member literal, raw as E.A
enum E {
	A,
	B,
	C,
}
function describe(value: E): string {
	switch (value) {
		case E.A:
			return "a";
		case E.B:
			return "b";
		case E.C:
			return "c";
		default: {
			const unreachable: never = value;
			return "unreachable";
		}
	}
}
const raw: number = 7;
console.log("before");
const tag: E.A = raw as E.A;
console.log(describe(tag));
console.log("after");
