package main

import (
	"context"
	"fmt"
	"github.com/system-inc/adamic/internal/load"
	"github.com/system-inc/adamic/internal/lower"
	"github.com/system-inc/adamic/internal/native"
	"github.com/system-inc/adamic/stage1/cohere/lint/registry"
	"os"
	"runtime/pprof"
	"time"
)

func main() {
	if _, err := registry.Generate("stage1/cohere/lint"); err != nil {
		panic(err)
	}
	start := time.Now()
	p, err := load.Load([]string{"stage1/cohere/lint/main.ts"})
	if err != nil {
		panic(err)
	}
	fmt.Println("load", time.Since(start))
	start = time.Now()
	ir, err := lower.Lower(context.Background(), p)
	if err != nil {
		panic(err)
	}
	fmt.Println("lower", time.Since(start))
	f, err := os.Create("review/compiler/c-emission-profile/lint.pb.gz")
	if err != nil {
		panic(err)
	}
	if err = pprof.StartCPUProfile(f); err != nil {
		panic(err)
	}
	start = time.Now()
	c := native.C(ir)
	elapsed := time.Since(start)
	pprof.StopCPUProfile()
	f.Close()
	fmt.Println("native.C", elapsed, "bytes", len(c))
	if err = os.WriteFile("/tmp/c-emission-lint.c", []byte(c), 0644); err != nil {
		panic(err)
	}
}
