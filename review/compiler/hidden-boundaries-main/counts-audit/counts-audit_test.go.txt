func auditClosureCounts(t *testing.T,name string,before,after string) { t.Helper(); path:="internal/oracle/testdata/"+name+".a"; want:=after; if os.Getenv("ADAMIC_COUNTS_AUDIT_BASELINE")=="1" { want=before }; if got:=counted(t,path,false,nil,false,false); got!=want { t.Fatalf("count attribution: got %s, want %s",got,want) }; absolute,pathErr:=filepath.Abs(filepath.Join(repository,path));if pathErr!=nil{t.Fatal(pathErr)}; truth:=onNode(t,absolute); program,err:=lowered(t,absolute);if err!=nil {t.Fatal(err)}; compiled,sanitized:=natively(t,program); js:=onJavaScriptBackend(t,program); for backend,got:=range map[string]run{"native":compiled,"javascript":js} { if name=="user_iterators_rest_tdz" {if string(got.stdout)!=string(truth.stdout)||got.exitCode!=70||truth.exitCode==0 {t.Fatalf("TDZ %s: got %+v, Node %+v",backend,got,truth)}} else if diff:=disagreement(truth,got);diff!="" {t.Fatalf("%s: %s",backend,diff)} }; if name!="user_iterators_rest_tdz" {if report:=leaks(t,program,sanitized);report!=""{t.Fatal(report)}};t.Logf("%s; Node stdout equal in both backends",want) }
