package oracle
import "testing"
func TestMovedCounts_lane1_interfaces_good(t *testing.T) { t.Parallel(); program,path := interfaceFixture(t,"lane1/interfaces-good"); want := onNode(t,path); if want.exitCode != 0 { t.Fatal(want) }; sanitized, leaked, comparable := nativelyCheckingLeaks(t,program,true); if leaked != "" || !comparable { t.Fatalf("leaks %s comparable %v", leaked, comparable) }; for lane,got := range map[string]run{"sanitized":sanitized,"native":released(t,program),"javascript":onJavaScriptBackend(t,program)} { t.Logf("%s exit=%d stdout=%q stderr=%q",lane,got.exitCode,got.stdout,got.stderr); if diff:=disagreement(want,got); diff!="" { t.Errorf("%s: %s",lane,diff) } } }
