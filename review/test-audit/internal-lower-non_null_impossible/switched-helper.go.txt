package lower

import "os"

func auditU040Selected(id string) bool { return os.Getenv("ADAMIC_MUTANT") == id }
func auditU040Bool(id string, original, changed func() bool) bool {
	if auditU040Selected(id) {
		return changed()
	}
	return original()
}
func auditU040String(id, original, changed string) string {
	if auditU040Selected(id) {
		return changed
	}
	return original
}
