package native
import "os"
func auditUnitMutant()string{return os.Getenv("ADAMIC_MUTANT")}
func auditUnitBool(id string,b bool)bool{if auditUnitMutant()==id{return !b};return b}
func auditUnitString(id,a,b string)string{if auditUnitMutant()==id{return a};return b}
func auditUnitInt(id string,a,b int)int{if auditUnitMutant()==id{return a};return b}
