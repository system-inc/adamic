25: // firstDifference says where two outputs first differ, line by line, or "" when they don't.

27: 	if got == want {
28: 		return ""

27: // Source bytes approximate parser and printer work. Ties use the original case
28: // number and then the shard number, making the assignment reproducible.
29: func assignCorpus(weights []int, count int) [][]int {
30: 	count = min(count, len(weights))
31: 	if count == 0 {
32: 		return nil
33: 	}
34: 	order := make([]int, len(weights))
35: 	for i := range order {
36: 		order[i] = i
37: 	}
38: 	sort.Slice(order, func(i, j int) bool {
39: 		if weights[order[i]] != weights[order[j]] {
40: 			return weights[order[i]] > weights[order[j]]
41: 		}
42: 		return order[i] < order[j]
43: 	})
44: 	result, totals := make([][]int, count), make([]int, count)
45: 	for _, index := range order {
46: 		shard := 0
47: 		for i := 1; i < count; i++ {
48: 			if totals[i] < totals[shard] {
49: 				shard = i
50: 			}
51: 		}
52: 		result[shard] = append(result[shard], index)
53: 		totals[shard] += max(weights[index], 1)
54: 	}
55: 	return result
56: }

134: func mergeCorpus(plan *corpusPlan, outputs [][]byte) ([]byte, error) {
135: 	if len(outputs) != len(plan.shards) {
136: 		return nil, fmt.Errorf("corpus: %d shards, want %d", len(outputs), len(plan.shards))
137: 	}
138: 	merged := make([][]byte, len(plan.labels))
139: 	for number, shard := range plan.shards {
140: 		output := outputs[number]
141: 		lines := bytes.Split(bytes.TrimSuffix(output, []byte("\n")), []byte("\n"))
142: 		if len(output) == 0 {
143: 			lines = nil
144: 		}
145: 		if len(lines) != len(shard.indices) {
146: 			index := shard.indices[min(len(lines), len(shard.indices)-1)]
147: 			return nil, fmt.Errorf("shard %d %s: missing or extra case: %d answers, want %d", number, plan.labels[index], len(lines), len(shard.indices))
148: 		}
149: 		if !bytes.HasSuffix(output, []byte("\n")) {
150: 			return nil, fmt.Errorf("shard %d %s: unterminated answer", number, plan.labels[shard.indices[len(shard.indices)-1]])
151: 		}
152: 		for i, index := range shard.indices {
153: 			if index < 0 || index >= len(merged) {
154: 				return nil, fmt.Errorf("shard %d: extra case %d", number, index)
155: 			}
156: 			if merged[index] != nil {
157: 				return nil, fmt.Errorf("shard %d %s: repeated case", number, plan.labels[index])
158: 			}
159: 			merged[index] = lines[i]
160: 		}
161: 	}
162: 	var result bytes.Buffer
163: 	for index, line := range merged {
164: 		if line == nil {
165: 			return nil, fmt.Errorf("%s: missing case", plan.labels[index])
166: 		}
167: 		result.Write(line)
168: 		result.WriteByte('\n')
169: 	}
170: 	return result.Bytes(), nil
171: }
