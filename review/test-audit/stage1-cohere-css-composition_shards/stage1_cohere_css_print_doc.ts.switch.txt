import { auditSelection } from './u078_selection.ts';
// The CSS-used subset of cohere's document IR and printer. Links are arena handles.
import { panic } from 'adamic';
import { repeatText } from './convert.ts';
import { wideStarts, wideEnds, emojiPatternSource, narrowEmojiPatternSource } from './print_width_data.ts';
const emojis = new RegExp(emojiPatternSource, 'gu');
const narrow = new RegExp(narrowEmojiPatternSource, 'u');
export function stringWidth(text: string): number {
    let ascii = true;
    for(let i = 0; i < text.length; i++) {
        const c = text.charCodeAt(i);
        if(c < 0x20 || c > 0x7f) { ascii = false; break; }
    }
    if(ascii) return text.length;
    let mapped = '';
    for(let i = 0; i < text.length; i++) {
        const c = text.charCodeAt(i);
        mapped += String.fromCodePoint(c >= 0xd800 && c <= 0xdfff ? 0x100000 + c - 0xd800 : c);
    }
    let width = 0;
    let rest = ''; let start = 0; emojis.lastIndex = 0;
    let match = emojis.exec(mapped);
    while(match !== null) {
        const text = match[0] ?? '';
        width += narrow.test(text) ? 1 : 2;
        rest += mapped.slice(start, match.index);
        start = match.index + text.length;
        match = emojis.exec(mapped);
    }
    rest += mapped.slice(start);
    let decoded = '';
    for(const c of rest) {
        const n = c.codePointAt(0) ?? 0;
        decoded += String.fromCharCode(n >= 0x100000 && n <= 0x1007ff ? n - 0x100000 + 0xd800 : n);
    }
    for(const c of decoded) {
        const n = c.codePointAt(0) ?? 0;
        if(n <= 31 || (n >= 127 && n <= 159) || (n >= 0x300 && n <= 0x36f) || (n >= 0xfe00 && n <= 0xfe0f)) continue;
        let lo = 0;
        let hi = wideEnds.length;
        while(lo < hi) { const mid = Math.floor((lo + hi) / 2); if((wideEnds[mid] ?? 0) < n) lo = mid + 1; else hi = mid; }
        width += lo < wideEnds.length && (wideStarts[lo] ?? 0) <= n ? 2 : 1;
    }
    return width;
}
export class DocNode {
    readonly kind: string;
    readonly parts: number[];
    readonly text: string;
    broken: boolean;
    private cachedWidth = -1;
    constructor(kind: string, parts: number[], text: string = '', broken: boolean = false) {
        this.kind = kind; this.parts = parts; this.text = text; this.broken = broken;
    }
    width(): number {
        if(this.cachedWidth < 0) this.cachedWidth = stringWidth(this.text);
        return this.cachedWidth;
    }
}
type Command = { readonly doc: number; readonly indent: number; readonly flat: boolean; readonly offset: number };
function command(doc: number, indent: number, flat: boolean, offset: number = 0): Command { return {doc, indent, flat, offset}; }
function emptyParts(): number[] { const parts: number[] = []; return parts; }
function emptyCommands(): Command[] { const commands: Command[] = []; return commands; }
export class Documents {
    readonly nodes: DocNode[] = [];
    add(kind: string, parts: number[] = emptyParts(), text: string = '', broken: boolean = false): number {
        const index = this.nodes.length; this.nodes.push(new DocNode(kind, parts, text, broken)); return index;
    }
    at(index: number): DocNode { return this.nodes[index] ?? panic('missing document'); }
    text(value: string): number { return this.add('text', emptyParts(), value); }
    concat(parts: number[]): number { return this.add('concat', parts); }
    group(content: number, broken: boolean = false): number { return this.add('group', [content], '', broken); }
    indent(content: number): number { return this.add('indent', [content]); }
    dedent(content: number): number { return this.add('dedent', [content]); }
    fill(parts: number[]): number { return this.add('fill', parts); }
    line(): number { return this.add('line'); }
    soft(): number { return this.add('soft'); }
    hard(): number { return this.concat([this.add('hard'), this.add('break')]); }
    literal(): number { return this.add('literal'); }
    ifBreak(content: number): number { return this.add('if', [content, this.text('')]); }
    suffix(content: number): number { return this.add('suffix', [content]); }
    boundary(): number { return this.add('boundary'); }
    join(separator: number, parts: number[]): number {
        const joined: number[] = [];
        for(const part of parts) { if(joined.length > 0) joined.push(separator); joined.push(part); }
        return this.concat(joined);
    }
    removeLines(index: number): number {
        const n = this.at(index);
        if(n.kind === 'line') return this.text(' ');
        if(n.kind === 'soft' || n.kind === 'if') return n.kind === 'if' ? this.removeLines(n.parts[1] ?? 0) : this.text('');
        const parts: number[] = emptyParts(); for(const p of n.parts) parts.push(this.removeLines(p));
        return this.add(n.kind, parts, n.text, n.broken);
    }
    propagate(index: number): boolean {
        const n = this.at(index);
        let broken = n.kind === 'break';
        for(const p of n.parts) { if(this.propagate(p)) broken = true; }
        if(n.kind === 'group') { n.broken = n.broken || broken; return n.broken; }
        return broken;
    }
    fits(next: Command, rest: readonly Command[], remaining: number, suffix: boolean, mustFlat: boolean): boolean {
        const stack: Command[] = [next]; let ri = rest.length; let pending = false;
        while(remaining >= 0) {
            if(stack.length === 0) { if(ri === 0) return true; ri--; stack.push(rest[ri] ?? panic('fit rest')); continue; }
            const c = stack.pop() ?? panic('fit command'); const n = this.at(c.doc);
            if(n.kind === 'text') {
                if(n.text !== '') { if(pending) { remaining--; pending = false; }
                    remaining -= n.width();
                }
            } else if(n.kind === 'group') {
                if(mustFlat && n.broken) return false;
                stack.push(command(n.parts[0] ?? 0, 0, c.flat && !n.broken));
            } else if(n.kind === 'line' || n.kind === 'soft' || n.kind === 'hard' || n.kind === 'literal') {
                if(!c.flat || n.kind === 'hard' || n.kind === 'literal') return true;
                if(n.kind === 'line') pending = true;
            } else if(n.kind === 'suffix') suffix = true;
            else if(n.kind === 'boundary') { if(suffix) return false; }
            else if(n.kind === 'if') stack.push(command(n.parts[c.flat ? 1 : 0] ?? 0, 0, c.flat));
            else if(n.kind !== 'break') {
                for(let i = n.parts.length - 1; i >= (n.kind === 'fill' ? c.offset : 0); i--) stack.push(command(n.parts[i] ?? 0, 0, c.flat));
            }
        }
        return false;
    }
    print(root: number, width: number, tabWidth: number, tabs: boolean): string {
        this.propagate(root);
        const stack: Command[] = [command(root, 0, false)]; const suffix: Command[] = [];
        let result = ''; let column = 0; let remeasure = false;
        while(stack.length > 0) {
            const c = stack.pop() ?? panic('print command'); const n = this.at(c.doc);
            if(n.kind === 'text') { result += n.text; if(stack.length > 0) column += n.width(); }
            else if(n.kind === 'indent' || n.kind === 'dedent') stack.push(command(n.parts[0] ?? 0, Math.max(0, c.indent + (n.kind === 'indent' ? (auditSelection === 'M4' ? 0 : 1) : -1)), c.flat));
            else if(n.kind === 'group') {
                let flat = c.flat && !remeasure && !n.broken;
                if(!c.flat || remeasure) { remeasure = false; flat = !n.broken && this.fits(command(n.parts[0] ?? 0, c.indent, true), stack, width - column, suffix.length > 0, false); }
                stack.push(command(n.parts[0] ?? 0, c.indent, flat));
            } else if(n.kind === 'fill') {
                const offset = c.offset; const length = n.parts.length - offset;
                if(length > 0) {
                    const content = n.parts[offset] ?? 0;
                    const fits = this.fits(command(content, c.indent, true), emptyCommands(), width - column, suffix.length > 0, true);
                    if(length > 2) {
                        const both = this.concat([content, n.parts[offset + 1] ?? 0, n.parts[offset + 2] ?? 0]);
                        const all = this.fits(command(both, c.indent, true), emptyCommands(), width - column, suffix.length > 0, true);
                        stack.push(command(c.doc, c.indent, c.flat, offset + 2));
                        stack.push(command(n.parts[offset + 1] ?? 0, c.indent, all));
                        stack.push(command(content, c.indent, fits));
                    } else { if(length === 2) stack.push(command(n.parts[offset + 1] ?? 0, c.indent, fits)); stack.push(command(content, c.indent, fits)); }
                }
            } else if(n.kind === 'if') stack.push(command(n.parts[c.flat ? 1 : 0] ?? 0, c.indent, c.flat));
            else if(n.kind === 'suffix') suffix.push(command(n.parts[0] ?? 0, c.indent, c.flat));
            else if(n.kind === 'boundary') { if(suffix.length > 0) stack.push(command(this.add('hard'), c.indent, c.flat)); }
            else if(n.kind === 'line' || n.kind === 'soft' || n.kind === 'hard' || n.kind === 'literal') {
                if(c.flat && (n.kind === 'line' || n.kind === 'soft')) { if(n.kind === 'line') { result += ' '; column++; } }
                else {
                    if(c.flat) remeasure = true;
                    if(suffix.length > 0) { stack.push(c); for(let i = suffix.length - 1; i >= 0; i--) stack.push(suffix[i] ?? panic('suffix')); suffix.splice(0); }
                    else if(n.kind === 'literal') { result += '\n'; column = 0; }
                    else { let end = result.length;
                        while(end > 0 && (result.charCodeAt(end - 1) === 32 || result.charCodeAt(end - 1) === 9)) end--;
                        // Concatenate directly after trimming: appending to a shared slice currently
                        // underflows the runtime's capacity check (gaps/7_shared_slice_append.ts).
                        const newline = '\n' + repeatText(tabs ? '\t' : ' ', c.indent * (tabs ? 1 : tabWidth));
                        if(end !== result.length) result = result.slice(0, end) + newline;
                        else result += newline; column = c.indent * tabWidth; }
                }
            } else if(n.kind !== 'break') { for(let i = n.parts.length - 1; i >= 0; i--) stack.push(command(n.parts[i] ?? 0, c.indent, c.flat)); }
            if(stack.length === 0 && suffix.length > 0) { for(let i = suffix.length - 1; i >= 0; i--) stack.push(suffix[i] ?? panic('suffix')); suffix.splice(0); }
        }
        return result;
    }
}
