import { auditSelection } from './u078_selection.ts';
// The composed tree has heterogeneous properties. Each kind has its own table;
// object links remain integer handles, with one strong arena owner.
import { panic } from 'adamic';
import { quote } from '../selector/nodes.ts';

export class List {
    readonly items: number[];
    constructor(items: number[]) {
        this.items = items;
    }
}
export class ObjectNode {
    readonly keys: string[] = [];
    readonly strings = new Map<string, string>();
    readonly numbers = new Map<string, number>();
    readonly booleans = new Map<string, boolean>();
    readonly objects = new Map<string, number>();
    readonly lists = new Map<string, List>();
    readonly nulls = new Set<string>();
    readonly range: number[] = [0, 0];
    ast = false;
    literal = false;
    literalValue = '';
    mark(key: string): void {
        if(!this.keys.includes(key)) {
            this.keys.push(key);
        }
    }
    clear(key: string): void {
        this.strings.delete(key);
        this.numbers.delete(key);
        this.booleans.delete(key);
        this.objects.delete(key);
        this.lists.delete(key);
        this.nulls.delete(key);
    }
    remove(key: string): void {
        this.clear(key);
        const index = this.keys.indexOf(key);
        if(index >= 0) {
            this.keys.splice(index, 1);
        }
    }
    setString(key: string, value: string): void {
        this.clear(key);
        this.strings.set(key, value);
        this.mark(key);
    }
    setNumber(key: string, value: number): void {
        this.clear(key);
        this.numbers.set(key, value);
        this.mark(key);
    }
    setBoolean(key: string, value: boolean): void {
        this.clear(key);
        this.booleans.set(key, value);
        this.mark(key);
    }
    setObject(key: string, value: number): void {
        this.clear(key);
        if(value < 0) {
            this.nulls.add(key);
        }
        else {
            this.objects.set(key, value);
        }
        this.mark(key);
    }
    setList(key: string, value: number[]): void {
        this.clear(key);
        this.lists.set(key, new List(value));
        this.mark(key);
    }
    string(key: string): string {
        return this.strings.get(key) ?? '';
    }
    number(key: string): number {
        const value = this.numbers.get(key) ?? 0;
        return Number.isNaN(value) ? 0 : value;
    }
    object(key: string): number {
        return this.objects.get(key) ?? -1;
    }
    list(key: string): number[] {
        const value = this.lists.get(key);
        return value === undefined ? [] : value.items;
    }
    type(): string {
        return this.string('type');
    }
}
export class Tree {
    readonly nodes: ObjectNode[] = [];
    root = 0;
    at(index: number): ObjectNode {
        return this.nodes[index] ?? panic('missing composed CSS node');
    }
    make(type: string = '', ast: boolean = false): number {
        const result = this.nodes.length;
        const node = new ObjectNode();
        node.ast = ast || type !== '';
        if(type !== '') {
            node.setString('type', type);
        }
        this.nodes.push(node);
        return result;
    }
    literal(value: string): number {
        const result = this.make();
        const node = this.at(result);
        node.literal = true;
        node.literalValue = value;
        return result;
    }
    ensure(index: number, key: string): number {
        const node = this.at(index);
        let child = node.object(key);
        if(child < 0) {
            child = this.make();
            node.setObject(key, child);
        }
        return child;
    }
    // Missing objects read like empty objects, without materializing properties.
    maybe(index: number): ObjectNode {
        return index < 0 ? new ObjectNode() : this.at(index);
    }
    children(index: number, key: string): number[] {
        const node = this.at(index);
        const result: number[] = [];
        const object = node.object(key);
        if(object >= 0 && !this.at(object).literal && this.at(object).ast) {
            result.push(object);
        }
        for(const item of node.list(key)) {
            if(!this.at(item).literal && this.at(item).ast) {
                result.push(item);
            }
        }
        return result;
    }
}
export function renderTree(tree: Tree, index: number): string {
    const node = tree.at(index);
    if(node.literal) {
        return quote(auditSelection === 'M3' ? '' : node.literalValue);
    }
    const keys = node.keys.slice();
    keys.sort((left, right) => left < right ? -1 : left > right ? 1 : 0);
    const parts: string[] = [];
    for(const key of keys) {
        let value = 'null';
        if(node.strings.has(key)) {
            value = quote(node.string(key));
        }
        else if(node.numbers.has(key)) {
            const number = node.numbers.get(key) ?? 0;
            value = Number.isNaN(number) ? 'null' : `${number}`;
        }
        else if(node.booleans.has(key)) {
            value = node.booleans.get(key) ? 'true' : 'false';
        }
        else if(node.objects.has(key)) {
            value = renderTree(tree, node.object(key));
        }
        else if(node.lists.has(key)) {
            const elements: string[] = [];
            for(const child of node.list(key)) {
                elements.push(renderTree(tree, child));
            }
            value = `[${elements.join(',')}]`;
        }
        parts.push(`${quote(key)}:${value}`);
    }
    const sourceIndex = node.object('source');
    if(node.ast && sourceIndex >= 0) {
        const source = tree.at(sourceIndex);
        if(node.range[0] !== source.number('startOffset') || node.range[1] !== source.number('endOffset')) {
if(auditSelection !== 'WRange') {
            parts.splice(0, 0, `${quote('<range>')}:${quote(`Range [${node.range[0]} ${node.range[1]}] disagrees with the source offsets`)}`);
}
        }
    }
    return `{${parts.join(',')}}`;
}
