import { auditSelection } from './u078_selection.ts';
// Parent and child edges are indexes. The arena is the only strong owner.
import { panic } from 'adamic';
import { quote } from '../selector/nodes.ts';
import type { Input, Position } from './input.ts';

export class RawValue {
    raw: string;
    readonly value: string;
    scss: string | undefined = undefined;
    constructor(raw: string, value: string) {
        this.raw = raw;
        this.value = value;
    }
}
export class CssNode {
    type: string;
    readonly keys: string[] = ['raws'];
    readonly rawStrings = new Map<string, string>();
    readonly rawBooleans = new Map<string, boolean>();
    readonly rawValues = new Map<string, RawValue>();
    readonly strings = new Map<string, string>();
    readonly booleans = new Map<string, boolean>();
    readonly children: number[] = [];
    parent = -1;
    readonly range: number[] = [0, 0];
    start: Position | undefined = undefined;
    end: Position | undefined = undefined;
    constructor(type: string) {
        this.type = type;
        if(type === 'root' || type === 'rule') {
            this.keys.push('nodes');
        }
    }
    mark(key: string): void {
        if(!this.keys.includes(key)) {
            this.keys.push(key);
        }
    }
    set(key: string, value: string): void {
        this.strings.set(key, value);
        this.mark(key);
    }
    get(key: string): string {
        return this.strings.get(key) ?? '';
    }
    raw(key: string): string {
        return this.rawStrings.get(key) ?? '';
    }
}
function positionJSON(position: Position, input: Input): string {
    return `{"column":${position.column},"line":${position.line},"offset":${position.offset}}`;
}
function objectJSON(fields: Map<string, string>): string {
    const keys: string[] = [];
    for(const key of fields.keys()) {
        keys.push(key);
    }
    keys.sort((left, right) => left < right ? -1 : left > right ? 1 : 0);
    const parts: string[] = [];
    for(const key of keys) {
        parts.push(`${quote(key)}:${fields.get(key) ?? panic('missing canonical field')}`);
    }
    return `{${parts.join(',')}}`;
}
export function render(nodes: readonly CssNode[], index: number, input: Input): string {
    const node = nodes[index] ?? panic('missing CSS node');
    const fields = new Map<string, string>();
    const raws = new Map<string, string>();
    for(const [key, value] of node.rawStrings) {
        raws.set(key, quote(value));
    }
    for(const [key, value] of node.rawBooleans) {
        raws.set(key, value ? 'true' : 'false');
    }
    for(const [key, value] of node.rawValues) {
        raws.set(key, `{"raw":${quote(value.raw)},${value.scss === undefined ? '' : `"scss":${quote(value.scss)},`}"value":${quote(value.value)}}`);
    }
    fields.set('raws', objectJSON(raws));
    fields.set('type', quote(node.type));
    const keys: string[] = [];
    for(const key of node.keys) {
        keys.push(quote(key));
        if(key === 'nodes') {
            const children: string[] = [];
            for(const child of node.children) {
                children.push(render(nodes, child, input));
            }
            fields.set(key, `[${children.join(',')}]`);
        }
        else if(key === 'source') {
            const start = node.start ?? panic('missing source start');
            fields.set(key, `{${node.end === undefined ? '' : `"end":${positionJSON(node.end, input)},`}"start":${positionJSON(start, input)}}`);
        }
        else if(node.strings.has(key)) {
            fields.set(key, quote(node.get(key)));
        }
        else if(node.booleans.has(key)) {
            fields.set(key, node.booleans.get(key) ? 'true' : 'false');
        }
    }
    if(node.start !== undefined && node.end !== undefined && (node.range[0] !== node.start.offset || node.range[1] !== node.end.offset)) {
if(auditSelection !== 'WRange') {
        fields.set('range', quote(`Range [${node.range[0]} ${node.range[1]}] is not source's [${node.start.offset} ${node.end.offset}]`));
}
    }
    fields.set('keys', `[${keys.join(',')}]`);
    return objectJSON(fields);
}
