func TestU012LimitObservation(t *testing.T) {
 e, err := prepareMode("../..", "testdata/mini", t.TempDir(), nil, true)
 if err != nil { t.Fatal(err) }
 e.cache = &resultCache{directory: t.TempDir()}
 e.inProcess = true
 e.jobs = 1
 e.log = io.Discard
 report, err := e.runFilter("", 1, false)
 if err != nil { t.Fatal(err) }
 t.Logf("total=%d skipped=%d unrun=%d attempted=%d", report.Total, report.Skipped, report.Unrun, report.Total-report.Skipped-report.Unrun)
}
