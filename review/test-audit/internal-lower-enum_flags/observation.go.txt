package lower
import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "path/filepath"
 "testing"
 "github.com/microsoft/TypeScript/tsc/shim/ast"
 "github.com/system-inc/adamic/internal/load"
 "github.com/system-inc/adamic/internal/ir"
)
// Observational harness only: it prints actual production results without an expected answer.
func TestAuditObservation(t *testing.T) {
 mode:=os.Getenv("AUDIT_OBSERVATION")
 source:="enum E {A=1,B=2,Alias=A} const x:E=E.A;"
 if mode=="flags" { source="enum Flags {None=0,A=1<<0,High=1<<30}" }
 if mode=="open" { source="enum E {A}" }
 if mode=="default-proof" { source="enum E {A,B} const e:E=E.A; switch(e){case E.A:break;default:{const n:never=e;}}" }
 if mode=="domain" { source="enum Flags {A=1<<0,B=1<<1} function f(a:Flags,n:number):Flags{return a & n;}" }
 if mode=="string-relation" { source="enum E {A='a',B='b'} const value = 'a';" }
 if mode=="member-relation" { source="enum E {A,B} function f(n:number):E.A{return E.A;}" }
 path:=filepath.Join(t.TempDir(),"observe.a");if err:=os.WriteFile(path,[]byte(source),0600);err!=nil {t.Fatal(err)}
 program,err:=load.Load([]string{path});if err!=nil {t.Fatal(err)}
 checked,release:=program.Checker(context.Background(),program.Files()[0]);defer release()
 l:=&lowering{program:program,checker:checked,result:&ir.Program{}}
 statements:=program.Files()[0].Statements.Nodes
 switch mode {
 case "values":
  result,err:=Lower(context.Background(),program);if err!=nil {fmt.Printf("OBS error=%v\n",err);return}; data,err:=json.Marshal(struct{Main []ir.Statement;Strings []string}{result.Main,result.Strings});if err!=nil {t.Fatal(err)};fmt.Printf("OBS %s\n",data)
 case "fields":
  fields,err:=l.enumFields(statements[0]);data,_:=json.Marshal(fields);fmt.Printf("OBS fields=%s error=%v\n",data,err)
 case "open":fmt.Printf("OBS openNumericEnumType=%v\n",l.openNumericEnumType(checked.GetTypeAtLocation(statements[0].Name())))
 case "default-proof":fmt.Printf("OBS enumDefaultUnreachable=%v\n",l.enumDefaultUnreachable(statements[2]))
 case "flags":fmt.Printf("OBS flagEnum=%v\n",l.flagEnum(l.symbol(statements[0].Name())))
 case "domain":
  fn:=statements[1].AsFunctionDeclaration();node:=fn.Body.AsBlock().Statements.Nodes[0].AsReturnStatement().Expression;fmt.Printf("OBS flagDomain=%v\n",l.flagDomain(node,l.symbol(statements[0].Name())))
 case "string-relation":
  node:=statements[1].AsVariableStatement().DeclarationList.AsVariableDeclarationList().Declarations.Nodes[0];fmt.Printf("OBS enumAssignable=%v\n",l.enumAssignable(checked.GetTypeAtLocation(node.Initializer()),checked.GetTypeAtLocation(statements[0].Name())))
 case "member-relation":
  fn:=statements[1].AsFunctionDeclaration();fmt.Printf("OBS enumAssignable=%v\n",l.enumAssignable(checked.GetTypeAtLocation(fn.Parameters.Nodes[0].Name()),checked.GetTypeAtLocation(fn.Body.AsBlock().Statements.Nodes[0].AsReturnStatement().Expression)))
 default:t.Fatal("unknown observation")
 }
 _=ast.KindUnknown
}
