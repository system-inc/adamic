package main
import("go/parser";"go/token";"go/ast";"strconv";"encoding/json";"os")
func main(){out:=map[string][]string{};for _,path:=range os.Args[1:] {if path=="--"{continue}; f,e:=parser.ParseFile(token.NewFileSet(),path,nil,0);if e!=nil{panic(e)};for _,decl:=range f.Decls {fn,ok:=decl.(*ast.FuncDecl);if !ok{continue};name:=fn.Name.Name;if name!="generated"&&name!="decoratedExports"&&name!="recoveredExpressions"{continue};ast.Inspect(fn.Body,func(n ast.Node)bool{lit,ok:=n.(*ast.BasicLit);if ok&&lit.Kind==token.STRING{s,e:=strconv.Unquote(lit.Value);if e!=nil{panic(e)};out[name]=append(out[name],s)};return true})}};json.NewEncoder(os.Stdout).Encode(out)}
