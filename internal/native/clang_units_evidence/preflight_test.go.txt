import("os";"path/filepath";"testing";"crypto/sha256";"fmt")
func TestUnitsMeasurementInputs(t *testing.T) { for _,name:=range []string{"lint-harness","typescript-parser","cohere-json","cohere-typeaware"} { base,err:=os.ReadFile(filepath.Join("/tmp/adamic-clang-evidence",name+".c"));if err!=nil {t.Fatal(err)}; changed,err:=os.ReadFile(filepath.Join("/tmp/adamic-clang-evidence",name+"-changed.c"));if err!=nil {t.Fatal(err)};h,u,err:=splitC(string(base));if err!=nil{t.Fatal(err)};h2,u2,err:=splitC(string(changed));if err!=nil{t.Fatal(err)};if h!=h2||len(u)!=len(u2){t.Fatalf("%s header or unit count differs",name)};count:=0;for i:=range u{if u[i]!=u2[i]{count++}};t.Logf("%s units=%d changed=%d header=%d",name,len(u),count,len(h)); hash:=sha256.New();fmt.Fprint(hash,h);for _,unit:=range u{fmt.Fprint(hash,unit.name,unit.source)};t.Logf("%s split fingerprint %x",name,hash.Sum(nil));if count!=1{t.Fatal("edit must change one unit")} } }
