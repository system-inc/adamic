package lower

// Measurement-only heap probe (task #rw2vjhn). Lives in the scratch overlay.
// HEAP_DIR: directory for heap-<N>GiB.pprof files. HEAP_SIDECAR: per-unit JSONL.
// The heap profile is the runtime's view as of the most recently completed GC
// cycle, and runtime.GC is never called here, so it shows what that cycle found live.

import (
	"encoding/json"
	"fmt"
	"github.com/microsoft/TypeScript/tsc/shim/ast"
	"github.com/microsoft/TypeScript/tsc/shim/checker"
	"os"
	"path/filepath"
	"runtime/metrics"
	"runtime/pprof"
	"sync"
	"time"
)

var heapSamples = []metrics.Sample{
	{Name: "/gc/heap/live:bytes"},
	{Name: "/memory/classes/heap/objects:bytes"},
	{Name: "/memory/classes/total:bytes"},
}
var heapMutex sync.Mutex
var heapSidecar *json.Encoder
var heapUnitCount int

func heapRead() (live, objects, total uint64) {
	heapMutex.Lock()
	defer heapMutex.Unlock()
	metrics.Read(heapSamples)
	return heapSamples[0].Value.Uint64(), heapSamples[1].Value.Uint64(), heapSamples[2].Value.Uint64()
}

func heapWriteProfile(name string) {
	directory := os.Getenv("HEAP_DIR")
	if directory == "" {
		return
	}
	file, err := os.Create(filepath.Join(directory, name))
	if err != nil {
		return
	}
	defer file.Close()
	_ = pprof.WriteHeapProfile(file)
}

func init() {
	if path := os.Getenv("HEAP_SIDECAR"); path != "" {
		if file, err := os.Create(path); err == nil {
			heapSidecar = json.NewEncoder(file)
		}
	}
	if os.Getenv("HEAP_DIR") == "" {
		return
	}
	go func() {
		next := uint64(1)
		for range time.Tick(250 * time.Millisecond) {
			live, objects, _ := heapRead()
			// Cross a new whole GiB by live-after-last-GC or by heap objects.
			level := objects >> 30
			if live>>30 > level {
				level = live >> 30
			}
			if level >= next {
				heapWriteProfile(fmt.Sprintf("heap-%dGiB.pprof", level))
				next = level + 1
			}
		}
	}()
}

// heapUnit records live heap after one unit finished.
func heapUnit(file, where string) {
	heapUnitCount++
	if heapSidecar == nil {
		return
	}
	live, objects, total := heapRead()
	_ = heapSidecar.Encode(map[string]any{"n": heapUnitCount, "file": file, "where": where,
		"live_after_last_gc": live, "heap_objects": objects, "runtime_total": total,
		"t": time.Now().UnixNano()})
}

func heapFinal() { heapWriteProfile("heap-exit.pprof"); memoStats() }

// Counting and optional memo of Checker.TypeToString for jsonDecodeSchema (task #rw2vjhn, scratch only).
// MEMO_TOSTRING=1 returns the first spelling of a type for every later call. The checker is
// single threaded here and one program has one checker, so a type pointer is a complete key.
type memoKey struct {
	checker *checker.Checker
	t       *checker.Type
}

var memoTable = map[memoKey]string{}
var memoCalls, memoMisses int
var memoDistinctSeen = map[memoKey]bool{}
var boundaryCalls int
var boundaryDistinctType = map[*checker.Type]bool{}
var boundaryDistinctNode = map[*ast.Node]bool{}
var boundaryDistinctPair = map[[2]any]bool{}

func (l *lowering) typeString(t *checker.Type) string {
	key := memoKey{l.checker, t}
	memoCalls++
	memoDistinctSeen[key] = true
	mode := os.Getenv("MEMO_TOSTRING")
	if mode == "1" || mode == "verify" || mode == "mutant" {
		if text, found := memoTable[key]; found {
			if mode == "verify" || mode == "mutant" {
				// verify: recompute and compare. mutant: the memo hands back a wrong spelling, to prove the check can fail.
				fresh := l.checker.TypeToString(t)
				if mode == "mutant" {
					text = "MUTANT"
				}
				memoVerified++
				if fresh != text {
					memoMismatches++
					fmt.Fprintf(os.Stderr, "MEMO MISMATCH: memo %q fresh %q\n", text, fresh)
				}
				if memoVerified%100000 == 0 {
					fmt.Fprintf(os.Stderr, "memo verified %d hits, %d mismatches\n", memoVerified, memoMismatches)
				}
			}
			return text
		}
		memoMisses++
		text := l.checker.TypeToString(t)
		memoTable[key] = text
		return text
	}
	return l.checker.TypeToString(t)
}

var memoVerified, memoMismatches int

func memoCountBoundary(node *ast.Node, proven *checker.Type) {
	boundaryCalls++
	boundaryDistinctType[proven] = true
	boundaryDistinctNode[node] = true
	boundaryDistinctPair[[2]any{node, proven}] = true
}

func memoStats() {
	path := os.Getenv("MEMO_STATS")
	if path == "" {
		return
	}
	file, err := os.Create(path)
	if err != nil {
		return
	}
	defer file.Close()
	_ = json.NewEncoder(file).Encode(map[string]any{
		"memo_enabled":                        os.Getenv("MEMO_TOSTRING") == "1",
		"jsonDecodeSchema_TypeToString_calls": memoCalls, "distinct_types": len(memoDistinctSeen), "memo_misses": memoMisses, "memo_hits_verified": memoVerified, "memo_mismatches": memoMismatches,
		"boundarySchema_calls": boundaryCalls, "boundary_distinct_proven_types": len(boundaryDistinctType),
		"boundary_distinct_declaration_nodes": len(boundaryDistinctNode), "boundary_distinct_node_type_pairs": len(boundaryDistinctPair),
		"units": heapUnitCount})
}
