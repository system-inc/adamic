function escapeJsonString(text: string): string {
    let result = '"';
    for (let index = 0; index < text.length; index++) {
        const code = text.charCodeAt(index);
        if (code === 34) result += '\\"';
        else if (code === 92) result += '\\\\';
        else if (code === 8) result += '\\b';
        else if (code === 9) result += '\\t';
        else if (code === 10) result += '\\n';
        else if (code === 12) result += '\\f';
        else if (code === 13) result += '\\r';
        else if (code >= 0xd800 && code <= 0xdbff) {
            if (index + 1 < text.length) {
                const next = text.charCodeAt(index + 1);
                if (next >= 0xdc00 && next <= 0xdfff) {
                    result += text.substring(index, index + 2);
                    index++;
                    continue;
                }
            }
            result += '\\u' + code.toString(16).padStart(4, '0');
        }
        else if (code < 32 || (code >= 0xdc00 && code <= 0xdfff)) {
            result += '\\u' + code.toString(16).padStart(4, '0');
        }
        else result += text.charAt(index);
    }
    return result + '"';
}

function escapeOptionalString(value: string | undefined): string {
    return value === undefined ? 'null' : escapeJsonString(value);
}

function formatDiagnosticArgument(value: string | number | undefined): string {
    if (typeof value === 'string') return escapeJsonString(value);
    if (typeof value === 'number') return Number.isFinite(value) ? String(value) : 'null';
    return 'null';
}
