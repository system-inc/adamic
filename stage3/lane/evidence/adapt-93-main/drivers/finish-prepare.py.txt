def run(name,cmd,limit=600,want=0,env=None):
 start=time.monotonic(); minimum=shutil.disk_usage(D).free
 with (D/(name+'.log')).open('w') as out:
  p=s.Popen(list(map(str,cmd)),cwd=R,env=env,stdout=out,stderr=s.STDOUT,start_new_session=True)

   minimum=min(minimum,shutil.disk_usage(D).free)
   if time.monotonic()-start>limit or minimum<3*1024**3:
