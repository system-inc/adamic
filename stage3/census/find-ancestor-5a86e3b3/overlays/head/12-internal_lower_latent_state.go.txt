// latentPrivateIRField is whether the full copy leaves an unexported IR field at its zero value.
func latentPrivateIRField(v reflect.Value, index int) bool {
	field := v.Type().Field(index).Name
	if v.Type() == reflect.TypeFor[ir.BoundaryStringTable]() && field == "indexes" {
		return true
	}
	if v.Type() == reflect.TypeFor[ir.Program]() && field == "argumentFacts" {
		if !v.Field(index).IsNil() {
			panic("latent state copy: populated IR cache Program.argumentFacts is unsupported")
		}
		return true
	}
	if v.Type() == reflect.TypeFor[ir.Program]() && field == "methodTables" {
		return true
	}
	return false
}

// latentSharedIRPointer is whether the full copy shares an IR pointer rather than copying what it points to.
func latentSharedIRPointer(t reflect.Type) bool {
	if t == reflect.TypeFor[*ir.GeneratedType]() {
		return true
	}
	return false
}
