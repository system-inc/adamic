package main
import("context";"fmt";"os";"github.com/system-inc/adamic/internal/load";"github.com/system-inc/adamic/internal/lower")
func main(){if len(os.Args)!=3{panic("usage: bind-seam entry.ts stream.jsonl")};os.Setenv("LATENT_FULL","0");os.Setenv("BIND_SEAM_STREAM",os.Args[2]);program,err:=load.LatentLoad([]string{os.Args[1]});if err!=nil{panic(err)};output,failure:=lower.LowerWithOptions(context.Background(),program,lower.Options{ModuleFormat:lower.ModuleFormatCommonJSBundle});if output!=nil||failure==nil||failure.Error()!="latent census: measurement only; no IR output"{panic("output guard failed")};exposed,failure:=load.Load([]string{os.Args[1]});if exposed!=nil||failure==nil{panic("loader output guard failed")};lower.BindSeamRun(context.Background(),program);fmt.Fprintln(os.Stderr,"seam finished",os.Args[1])}
