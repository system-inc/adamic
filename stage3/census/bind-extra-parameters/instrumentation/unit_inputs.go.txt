 locals:=[]map[string]any{};for i,x:=range l.result.Locals{locals=append(locals,map[string]any{"id":i,"name":x.Name,"function":x.Function,"global":x.Global})}

 row:=map[string]any{"unit":unit.Where,"locals":locals,"functions":functions,"binds":binds,"main_statements":len(l.result.Main),"function_type_target_keys":len(l.result.FunctionTypeTargets),"measurement":"isolated unit inputs only; no merge or bind judgment"}
 if err:=json.NewEncoder(out).Encode(row);err!=nil{panic(err)}
}
