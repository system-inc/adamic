// Finalized with the unit status and instantiation metadata, once all modes have
// returned their records. Scan findings alone are not body-lowering stops.
func latentAttachPassLedger(record *LatentFile) {
	for i := range record.Units {
		unit := &record.Units[i]
		switch unit.Status {
		case "attempted", "panic", "speculative":
		default:
			continue // A deferred or checker-excluded body was not attempted independently.
		}
		stopped := unit.Status == "panic"
		for _, finding := range record.Findings {
			if finding.Unit == unit.Where && (finding.Phase == "lowering" || finding.Phase == "deferred" || finding.Phase == "registration" || finding.Phase == "prepass") {
				stopped = true
			}
		}
		if stopped || len(unit.PassLedger) == 0 {
			unit.PassLedger = latent.OmittedPassLedger(latentPostLoweringSteps, stopped)
		}
	}
}
