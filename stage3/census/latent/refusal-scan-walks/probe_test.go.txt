type scanSite struct{ ChildWalks, FileWalks, Children int64 }
type scanMetric struct {
	Name                                   string
	Calls, ChildWalks, FileWalks, Children int64
	Seconds                                float64
	Sites                                  map[string]*scanSite
	Files                                  map[string]int64
}

var scanActive *scanMetric

func scanWalk(node *ast.Node, visit ast.Visitor, site string) bool {
	if scanActive == nil {
		return node.ForEachChild(visit)
	}
	metric := scanActive
	metric.ChildWalks++
	if metric.Sites == nil {
		metric.Sites = map[string]*scanSite{}
	}
	if metric.Sites[site] == nil {
		metric.Sites[site] = &scanSite{}
	}
	local := metric.Sites[site]
	local.ChildWalks++
	if node.Kind == ast.KindSourceFile {
		metric.FileWalks++
		local.FileWalks++
		if metric.Files == nil {
			metric.Files = map[string]int64{}
		}
		metric.Files[string(node.AsSourceFile().FileName())]++
	}
	return node.ForEachChild(func(child *ast.Node) bool { metric.Children++; local.Children++; return visit(child) })
}
func scanCPU() float64 {
	var r syscall.Rusage
	syscall.Getrusage(syscall.RUSAGE_SELF, &r)
	return float64(r.Utime.Sec+r.Stime.Sec) + float64(r.Utime.Usec+r.Stime.Usec)/1e6
}
