package main

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"github.com/system-inc/adamic/internal/load"
	"github.com/system-inc/adamic/internal/lower"
	"github.com/system-inc/adamic/stage3/census/moduleformat"
	"io"
	"os"
	"path/filepath"
	"sort"
)

func main() {
	if len(os.Args) != 3 {
		panic("usage: latent source-directory output.json")
	}
	var paths []string
	err := filepath.WalkDir(os.Args[1], func(path string, entry os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if !entry.IsDir() && (filepath.Ext(path) == ".ts" || filepath.Ext(path) == ".a") {
			paths = append(paths, path)
		}
		return nil
	})
	if err != nil {
		panic(err)
	}
	sort.Strings(paths)
	var witness io.Writer
	var witnessRows bytes.Buffer
	var witnessPath string
	if os.Getenv("LATENT_WITNESS") == "1" || os.Getenv("LATENT_MUTANT_WITNESS_DEFAULT") == "1" {
		witnessPath = os.Getenv("LATENT_WITNESS_OUT")
		if witnessPath == "" {
			witnessPath = os.Args[2] + ".witness.jsonl"
		}
		witness = &witnessRows
	}
	output, err := os.Create(os.Args[2])
	if err != nil {
		panic(err)
	}
	defer output.Close()
	encoder := json.NewEncoder(output)
	program, err := load.LatentLoad(paths)
	if err != nil {
		panic(err)
	}
	if os.Getenv("LATENT_ASSERT_NO_OUTPUT") != "" {
		output, failure := lower.LowerWithOptions(context.Background(), program, lower.Options{ModuleFormat: moduleformat.Resolve(lower.ModuleFormatCommonJSBundle, lower.ModuleFormatEcmaScript), ProgramRegion: os.Getenv("ADAMIC_PROGRAM_REGION") == "1", OwnershipQuery: os.Getenv("ADAMIC_OWNERSHIP_QUERY") == "1"})
		if output != nil || failure == nil || failure.Error() != "latent census: measurement only; no IR output" {
			panic("measurement returned usable IR")
		}
		outputProgram, failure := load.Load(paths)
		if outputProgram != nil || failure == nil {
			panic("measurement loader exposed an output program")
		}
	}
	diagnostics := program.LatentDiagnostics()
	if diagnostics == nil {
		diagnostics = []string{}
	}
	header := map[string]any{"module_format": moduleformat.Name(), "measurement": lower.LatentMeasurementLabel, "status": "measurement", "latent_mode": censusMode(), "checker_rejected": len(diagnostics) > 0, "diagnostics": diagnostics, "diagnostic_sites": program.LatentDiagnosticSites()}
	writeLatentStream(encoder, header, output.Sync, func(emit func(lower.LatentFile)) lower.LatentRelationReport {
		return lower.LatentWithWitness(context.Background(), program, func(record lower.LatentFile) {
			emit(record)
			fmt.Fprintln(os.Stderr, record.File, len(record.Units), len(record.Findings), lower.LatentMeasurementLabel)
		}, witness)
	})
	if witness != nil {
		// The census and timing streams already exist. Exclusive creation also
		// protects them if their paths resolve through symlinks or hard links.
		stream, err := os.OpenFile(witnessPath, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0666)
		if err != nil {
			panic(err)
		}
		_, writeErr := witnessRows.WriteTo(stream)
		closeErr := stream.Close()
		if writeErr != nil {
			panic(writeErr)
		}
		if closeErr != nil {
			panic(closeErr)
		}
	}
}

// Flush the header before any lowering. A cutoff cannot manufacture an end
// answer: the relation trailer is written only after the run returns normally.
func writeLatentStream(encoder *json.Encoder, header map[string]any, flush func() error, run func(func(lower.LatentFile)) lower.LatentRelationReport) {
	writeRecord := func(record any) {
		if err := encoder.Encode(record); err != nil {
			panic(err)
		}
		if err := flush(); err != nil {
			panic(err)
		}
	}
	writeRecord(header)
	relations := run(func(record lower.LatentFile) { writeRecord(record) })
	writeRecord(map[string]any{"module_format": moduleformat.Name(), "measurement": lower.LatentMeasurementLabel, "status": "relation_report", "relation_overflow": relations, "findings": relations.Findings})
}

func censusMode() string {
	if os.Getenv("LATENT_SPECULATIVE") == "1" && os.Getenv("LATENT_MUTANT_NO_STUBS") != "1" {
		return "speculative"
	}
	if os.Getenv("LATENT_FULL") == "1" {
		return "full"
	}
	return "first-error"
}
