from pathlib import Path
import re
import subprocess

paths = set(subprocess.check_output(['git','diff','--name-only','origin/main'],text=True).splitlines())
paths.update(subprocess.check_output(['git','ls-files','--others','--exclude-standard'],text=True).splitlines())
pattern = re.compile(r'(?<![A-Za-z0-9_./-])(?:[A-Za-z0-9_.-]+/)*[A-Za-z0-9_.-]+_test\.[A-Za-z0-9_.-]+')
violations = []
checked = 0
for name in sorted(paths):
    path = Path(name)
    if not path.is_file() or path.suffix in ['.txt','.md','.gz'] or re.search(r'(^|/)(test_[^/]+\.py|[^/]+_test\.go)$',name):
        continue
    try:
        text = path.read_text()
    except UnicodeDecodeError:
        continue
    checked += 1
    for line,body in enumerate(text.splitlines(),1):
        if pattern.search(body):
            violations.append(f'{name}:{line}: {body}')
print(f'Inspected {checked} changed non-test text files outside allowed documentation/log extensions.')
print('\n'.join(violations) if violations else 'PASS: no forbidden test-file references.')
raise SystemExit(bool(violations))
