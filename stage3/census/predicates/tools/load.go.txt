// Scratch measurement access only. Production Load remains inaccessible in this binary.
package load

import (
	"github.com/microsoft/TypeScript/tsc/shim/ast"
	"os"
)

func (p *Program) LatentDiagnostics() []string { return p.latentDiagnostics }

// Use byte ranges from the checker, not parsed display lines or string matching.
func (p *Program) LatentDiagnosticsIn(node *ast.Node) []string {
	if node == nil {
		return nil
	}
	file := ast.GetSourceFileOfNode(node)
	start := node.Pos()
	if os.Getenv("LATENT_MUTANT_BODY_SCOPE") != "" {
		start = node.Parent.Pos()
	}
	var result []string
	for _, diagnostic := range p.latentSites {
		if diagnostic.File() == file && diagnostic.Pos() < node.End() && (diagnostic.End() > start || diagnostic.Pos() >= start) {
			result = append(result, p.formatDiagnostic(diagnostic))
		}
	}
	return result
}

// Preserve raw checker spans so the independent recount can verify eligibility.
type LatentDiagnosticSite struct {
	Measurement string `json:"measurement"`
	File        string `json:"file"`
	Start       int    `json:"start"`
	End         int    `json:"end"`
	Text        string `json:"text"`
}

func (p *Program) LatentDiagnosticSites() []LatentDiagnosticSite {
	result := []LatentDiagnosticSite{}
	for _, diagnostic := range p.latentSites {
		file := ""
		if diagnostic.File() != nil {
			file = p.FileName(diagnostic.File())
		}
		result = append(result, LatentDiagnosticSite{Measurement: "measured on a checker-rejected program", File: file, Start: diagnostic.Pos(), End: diagnostic.End(), Text: p.formatDiagnostic(diagnostic)})
	}
	return result
}
