// Measurement only. Optional IR types are matched by name for main compatibility.
package lower
import("reflect";"os";"github.com/system-inc/adamic/internal/ir")
func priceCount(roots ...any) (elements, checks int) {
 seen:=map[any]bool{}
 var walk func(reflect.Value)
 walk=func(v reflect.Value){
  if !v.IsValid(){return}
  switch v.Kind(){
  case reflect.Interface: if !v.IsNil(){walk(v.Elem())};return
  case reflect.Pointer:
   if v.IsNil()||v.Type().Elem().PkgPath()!="github.com/system-inc/adamic/internal/ir"{return}
   key:=v.Interface();if seen[key]{return};seen[key]=true;walk(v.Elem());return
  case reflect.Struct:
   if v.Type().PkgPath()!="github.com/system-inc/adamic/internal/ir"{return}
   switch v.Type().Name(){
   case "ViewElement":if os.Getenv("V3_MUTANT_SKIP_NODE")!="ViewElement"{elements++}
   case "ViewElementCheck":if os.Getenv("V3_MUTANT_SKIP_NODE")!="ViewElementCheck"{checks++}
   }
   for i:=0;i<v.NumField();i++{if v.Type().Field(i).IsExported(){walk(v.Field(i))}}
  case reflect.Array,reflect.Slice:for i:=0;i<v.Len();i++{walk(v.Index(i))}
  }
 }
 for _,root:=range roots{walk(reflect.ValueOf(root))};return
}
func priceObserve(unit *LatentUnit,l *lowering,statements []ir.Statement){
 roots:=[]any{statements}
 for _,function:=range l.result.Functions{roots=append(roots,function.Body)}
 unit.ViewElements,unit.ViewElementChecks=priceCount(roots...);unit.IRMeasured=true
}
