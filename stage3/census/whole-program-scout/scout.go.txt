// Measurement overlay only. No IR crosses this API.
package lower
import (
 "context"
 "fmt"
 "os"
 "github.com/microsoft/TypeScript/tsc/shim/ast"
 "github.com/system-inc/adamic/internal/load"
)
type ScoutStop struct { Rule, Site, Words string }
type ScoutReport struct { Stops []ScoutStop; Completed bool; Functions, Locals int; Boundary string; UnderlyingAbort *ScoutStop }
var scoutPhase string
var scoutStops []ScoutStop
var scoutUnderlying *ScoutStop
func scoutDescribe(err error) ScoutStop {
 stop := ScoutStop{Rule:scoutPhase, Words:err.Error()}
 switch e := err.(type) { case *NotYet: stop.Site=e.Where; stop.Words=e.What; case *Refused: stop.Site=e.Where; stop.Words=e.What }
 return stop
}
func latentRecord(err error) { if err != nil { scoutStops=append(scoutStops,scoutDescribe(err)) } }
// Single invocation per process, as the census runner uses. Panic is a boundary,
// never evidence that a later pass ran or that missing edges are harmless.
func Scout(ctx context.Context, p *load.Program) (r ScoutReport) {
 scoutStops=nil; scoutUnderlying=nil; scoutPhase="entry"
 defer func(){ if e:=recover(); e!=nil { r.Boundary="panic: "+fmt.Sprint(e) }; r.Stops=scoutStops; r.UnderlyingAbort=scoutUnderlying }()
 result, err := scoutLowerWithOptions(ctx,p,Options{ModuleFormat:ModuleFormatCommonJSBundle})
 if err!=nil { latentRecord(err); r.Boundary="production abort at "+scoutPhase; return }
 r.Completed=true; r.Functions=len(result.Functions); r.Locals=len(result.Locals)
 return
}

func (l *lowering) scoutRefuse(m *ast.SourceFile) error { if os.Getenv("SCOUT_MUTANT_FIRST_ERROR")=="1" { return l.refuse(m) }; return l.latentRefuse(m) }
