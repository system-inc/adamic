// overloadSetValueStop stops an overload set read as a value. A call through the value resolves to
// the slot's own signature, never to an overload, so a narrow overload's result check would never
// run there (#z5d0jye, Compiler's must on 19ab1c58). functionValue asks it of a module-level set and
// nestedReference of a nested one, so both say the same words.
func (l *lowering) overloadSetValueStop(node *ast.Node, symbol *ast.Symbol) error {
	if symbol == nil {
		return nil
	}
	for _, declaration := range symbol.Declarations {
		if declaration.Kind == ast.KindFunctionDeclaration && declaration.Body() == nil && l.censusImplementation(declaration) != nil {
			return l.notYet(node, "an overloaded function as a value")
		}
	}
	return nil
}

func (l *lowering) nestedReference(node *ast.Node) (ir.Expression, bool, error) {
	local, ok := l.locals[l.symbol(node)]
	if !ok || l.result.Locals[local].NestedFunction == 0 {
		return nil, false, nil
	}
	if l.result.Locals[local].NestedFunction < 0 {
		return nil, true, l.notYet(node, "a generic function as a value")
	}
	if l.result.Functions[l.result.Locals[local].NestedFunction-1].Generator {
		// A call through the value would make a frame no consumer keeps to itself, as a module
		// generator's functionValue stops (#qk8rztp S3a). Its own calls take their carrier from
		// nestedGeneratorCall, never through here, so the words say what reaches here: anything but
		// its call by name (Lower's words on #qk8rztp S3c).
		return nil, true, l.generatorNotYet(node, "declared inside a function, used other than by a call to it by name")
	}
	if node != l.nestedCallee {
		if err := l.overloadSetValueStop(node, l.symbol(node)); err != nil {
			return nil, true, err
		}
	}
	return l.nestedValue(local), true, nil
}

// namespaceContainerLiteral is a namespace's object: each exported function's canonical value, in
// declaration order. namespaceValueAdmission has already stopped every other export that has a value.
func (l *lowering) namespaceContainerLiteral(declaration *ast.Node) (ir.Expression, error) {
	fields := []ir.Field{}
	for _, statement := range namespaceStatements(declaration) {
		if statement.Kind != ast.KindFunctionDeclaration || statement.Body() == nil || !ast.HasSyntacticModifier(statement, ast.ModifierFlagsExport) {
			continue
		}
		function, found := l.functions[l.symbol(statement.Name())]
		if !found {
			return nil, l.notYet(statement, "an exported function in a namespace used as a value, without a lowered function")
		}
		value, err := l.functionValue(statement.Name(), function)
		if err != nil {
			return nil, err
		}
		fields = append(fields, ir.Field{Name: statement.Name().Text(), Value: value})
	}
	return ir.ObjectLiteral{Fields: fields}, nil
}
