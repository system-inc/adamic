// Reduced from TypeScript 6.0.3 core.ts and factory/utilitiesPublic.ts.
// memoize keeps the returned function of T; this reduction evaluates the callback
// once before capture because current main refuses a captured callback cycle.
function memoize<T>(callback: () => T): () => T {
    const value = callback();
    return () => value;
}

function forEach<T, U>(array: readonly T[] | undefined, callback: (value: T) => U | undefined): U | undefined {
    if (array !== undefined) {
        for (const value of array) {
            const result = callback(value);
            if (result) {
                return result;
            }
        }
    }
    return undefined;
}

function filter<T>(array: readonly T[] | undefined, callback: (value: T) => boolean): readonly T[] | undefined {
    if (array === undefined) {
        return undefined;
    }
    const result: T[] = [];
    for (const value of array) {
        if (callback(value)) {
            result.push(value);
        }
    }
    return result;
}

interface TextRange {
    pos: number;
    end: number;
}
function setTextRange<T extends TextRange>(range: T, location: TextRange | undefined): T {
    if (location !== undefined) {
        range.pos = location.pos;
        range.end = location.end;
    }
    return range;
}

function append<T>(to: T[] | undefined, value: T | undefined): T[] | undefined {
    if (value === undefined) {
        return to;
    }
    if (to === undefined) {
        return [value];
    }
    to.push(value);
    return to;
}
