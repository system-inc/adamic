520: export function computeLineOfPosition(lineStarts: readonly number[], position: number, lowerBound?: number): number {
521:     let lineNumber = binarySearch(lineStarts, position, identity, compareValues, lowerBound);
522:     if (lineNumber < 0) {
523:         // If the actual position was not found,
524:         // the binary search returns the 2's-complement of the next line start
525:         // e.g. if the line starts at [5, 10, 23, 80] and the position requested was 20
526:         // then the search will return -2.
527:         //
528:         // We want the index of the previous line start, so we subtract 1.
529:         // Review 2's-complement if this is confusing.
530:         lineNumber = ~lineNumber - 1;
531:         Debug.assert(lineNumber !== -1, "position cannot precede the beginning of the file");
532:     }
533:     return lineNumber;
534: }

1238: export function getTokenPosOfNode(node: Node, sourceFile?: SourceFileLike, includeJsDoc?: boolean): number {
1239:     // With nodes that have no width (i.e. 'Missing' nodes), we actually *don't*
1240:     // want to skip trivia because this will launch us forward to the next token.
1241:     if (nodeIsMissing(node)) {
1242:         return node.pos;
1243:     }
1244: 
1245:     if (isJSDocNode(node) || node.kind === SyntaxKind.JsxText) {
1246:         // JsxText cannot actually contain comments, even though the scanner will think it sees comments
1247:         return skipTrivia((sourceFile ?? getSourceFileOfNode(node)).text, node.pos, /*stopAfterLineBreak*/ false, /*stopAtComments*/ true);
1248:     }
1249: 
1250:     if (includeJsDoc && hasJSDocNodes(node)) {
1251:         return getTokenPosOfNode(node.jsDoc![0], sourceFile);
1252:     }
1253: 
1254:     // For a syntax list, it is possible that one of its children has JSDocComment nodes, while
1255:     // the syntax list itself considers them as normal trivia. Therefore if we simply skip
1256:     // trivia for the list, we may have skipped the JSDocComment as well. So we should process its
1257:     // first child to determine the actual position of its first token.
1258:     if (node.kind === SyntaxKind.SyntaxList) {
1259:         sourceFile ??= getSourceFileOfNode(node);
1260:         const first = firstOrUndefined(getNodeChildren(node, sourceFile));
1261:         if (first) {
1262:             return getTokenPosOfNode(first, sourceFile, includeJsDoc);
1263:         }
1264:     }
1265: 
1266:     return skipTrivia(
1267:         (sourceFile ?? getSourceFileOfNode(node)).text,
1268:         node.pos,
1269:         /*stopAfterLineBreak*/ false,
1270:         /*stopAtComments*/ false,
1271:         isInJSDoc(node),
1272:     );
1273: }
