351: // namespaceValueArgument judges a namespace value passed to a call. It's kept only as an argument to a
352: // direct call of one program FunctionDeclaration, named by the callee identifier, with one declaration, a
353: // body and no type parameters, whose own parameter at that place is declared typeof N or typeof N |
354: // undefined with no type parameter in it. That's the body that runs, so every use of the value inside it is
355: // judged by this pass as its own. Anything else could run a body that sees the value as a wider type, where
356: // a write through it would lower unjudged: an overload (the chosen signature isn't the body), a function
357: // value, a method (an override or implementation may widen its parameter), a generic class's method or an
358: // interface reached through I<typeof N>, and a constructor, whose argument may become a field (Lower's and
359: // Compiler's musts, the rule for the third shape rather than a fourth case).
360: func (l *lowering) namespaceValueArgument(node, outer, call, declaration *ast.Node) error {
361: 	seen := func(what string) error {
362: 		return l.notYet(node, "a namespace value seen as "+what)
363: 	}
364: 	if call.Expression() == outer {
365: 		return seen("a callee")
366: 	}
367: 	if call.Kind == ast.KindNewExpression {
368: 		return seen("a constructor argument")
369: 	}
370: 	// A value a caller set a slot for (refuseInSlot) is written into a slot this can't follow, so it stops.
371: 	if l.slotSet(outer) {
372: 		return seen("a value written into a slot a caller set")
373: 	}
374: 	callee := ast.SkipParentheses(call.Expression())
375: 	if callee.Kind == ast.KindPropertyAccessExpression && l.isLibraryGlobal(callee.Expression(), "Object") {
376: 		return seen("an argument to Object." + callee.Name().Text())
377: 	}
378: 	function := l.namespaceDirectCallee(callee)
379: 	if function == nil {
380: 		return seen("an argument to a callee whose body can't be named")
381: 	}
382: 	declared := l.namespaceDeclaredParameter(call, outer, function)
383: 	if declared == nil {
384: 		return seen("an argument with no declared parameter of its own type")
385: 	}
