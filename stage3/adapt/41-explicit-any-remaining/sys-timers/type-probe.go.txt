package main
import("context";"encoding/json";"os";"path/filepath";"sort";"strings";"github.com/system-inc/adamic/internal/load";"github.com/microsoft/TypeScript/tsc/shim/ast";"github.com/microsoft/TypeScript/tsc/shim/checker")
func main(){root:=os.Args[1]; var roots []string; err:=filepath.WalkDir(filepath.Join(root,"src/compiler"),func(p string,e os.DirEntry,err error)error{if err!=nil{return err};if !e.IsDir()&&strings.HasSuffix(p,".ts"){roots=append(roots,p)};return nil});if err!=nil{panic(err)};sort.Strings(roots);if len(roots)!=79{panic("expected 79 roots")};p,err:=load.Load(roots);if err!=nil{panic(err)};enc:=json.NewEncoder(os.Stdout);for _,f:=range p.Files(){if filepath.Base(p.FileName(f))!="sys.ts"{continue};c,done:=p.Checker(context.Background(),f);var visit ast.Visitor;visit=func(n *ast.Node)bool{if ast.IsIdentifier(n)&&(n.Text()=="setTimeout"||n.Text()=="clearTimeout"){start:=int(n.Pos());end:=int(n.End());if start<0||end>len(f.Text()){panic("source span")};row:=map[string]any{"where":p.Where(n),"name":n.Text(),"type":c.TypeToString(c.GetTypeAtLocation(n)),"parent_kind":n.Parent.Kind.String(),"parent":f.Text()[n.Parent.Pos():n.Parent.End()]};if n.Parent.Kind==ast.KindShorthandPropertyAssignment{ct:=checker.Checker_getContextualType(c,n.Parent.Parent,0);if ct==nil{panic("missing timer contextual type")};property:=c.GetPropertyOfType(ct,n.Text());if property==nil{panic("missing timer contextual property")};row["destination"]=c.TypeToString(c.GetTypeOfSymbol(property))};var signatures []map[string]any;for _,sig:=range c.GetSignaturesOfType(c.GetTypeAtLocation(n),checker.SignatureKindCall){var params []string;for _,param:=range sig.Parameters(){params=append(params,c.TypeToString(c.GetTypeOfSymbol(param)))};signatures=append(signatures,map[string]any{"parameters":params,"return":c.TypeToString(c.GetReturnTypeOfSignature(sig))})};row["signatures"]=signatures;for parent:=n.Parent;parent!=nil;parent=parent.Parent{if parent.Kind==ast.KindFunctionDeclaration&&parent.Name()!=nil{row["owner"]=parent.Name().Text();break}};if err:=enc.Encode(row);err!=nil{panic(err)}};return n.ForEachChild(visit)};visit(f.AsNode());done()}}
