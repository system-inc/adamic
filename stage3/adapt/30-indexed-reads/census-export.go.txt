package main

import (
	"encoding/json"
	"errors"
	"os"
	"path/filepath"
	"testing"

	"github.com/system-inc/adamic/internal/load"
)

// The tree already contains the explicitly selected source adaptations.
// Do not invoke the meter's implicit optional-property adaptation.
func TestIndexedReadsCensus(t *testing.T) {
	var paths []string
	err := filepath.WalkDir(os.Getenv("ADAMIC_INDEXED_ROOT"), func(name string, entry os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if !entry.IsDir() && filepath.Ext(name) == ".ts" {
			paths = append(paths, name)
		}
		return nil
	})
	if err != nil {
		t.Fatal(err)
	}
	_, err = load.Load(paths)
	var diagnostics *load.CheckError
	if !errors.As(err, &diagnostics) {
		t.Fatalf("expected checker diagnostics, got %v", err)
	}
	data, err := json.MarshalIndent(struct {
		Roots       int      `json:"roots"`
		Diagnostics []string `json:"diagnostics"`
	}{len(paths), diagnostics.Diagnostics}, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("ADAMIC_INDEXED_OUT"), data, 0644); err != nil {
		t.Fatal(err)
	}
}
