interface Type { flags: number }
interface TypeParameter extends Type { constraint?: Type; isThisType?: boolean }
function isThisTypeParameter(type: Type): boolean {
	return !!(type.flags & 1 && (type as TypeParameter).isThisType);
}
const first: Type = { flags: 1 };
const second: Type = { flags: 2 };
console.log(String(isThisTypeParameter(first)) + " " + String(isThisTypeParameter(second)));
