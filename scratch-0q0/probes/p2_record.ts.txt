interface Options {
	readonly name?: string;
}
const table: Record<string, string> = {};
table["k"] = "v";
function show(value: Options | null | undefined): void {
	console.log(value?.name ?? "missing");
}
show(null);
show({ name: "set" });
show(undefined);
show(table);
console.log(String(Object.keys(table).length));
